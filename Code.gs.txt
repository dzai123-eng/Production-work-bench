function doGet(e) {
  const t = HtmlService.createTemplateFromFile('Index');
  const page = String((e && e.parameter && e.parameter.page) || 'monitor').toLowerCase();
  const sn = String((e && e.parameter && e.parameter.sn) || '').trim();
  const validPages = ['monitor', 'action', 'workbench', 'routing', 'component', 'chat'];
  const resolvedPage = validPages.indexOf(page) !== -1 ? page : 'monitor';

  t.initialPage = resolvedPage;
  t.initialSn = sn;

  const title = resolvedPage === 'action'
    ? 'Production Workbench · Action Center'
    : (resolvedPage === 'routing'
    ? 'Production Workbench · Routing'
    : (resolvedPage === 'workbench'
      ? 'Production Workbench · Mail'
      : (resolvedPage === 'component'
        ? 'Production Workbench · Component'
        : (resolvedPage === 'chat' ? 'Production Workbench · Chat' : APP_NAME))));

  return t
    .evaluate()
    .setTitle(title)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function buildHistorySourceSignature_(settings, plannerSheets, engineerSheets, productionSheets, ftzSheets) {
  function pack_(sheets) {
    return (sheets || []).map(function(sh) {
      return {
        spreadsheetId: sh.getParent().getId(),
        tab: sh.getName()
      };
    }).sort(function(a,b) {
      return (a.spreadsheetId + '|' + a.tab).localeCompare(b.spreadsheetId + '|' + b.tab);
    });
  }

  return JSON.stringify({
    settings: historySourceSettings_(settings),
    plannerSheets: pack_(plannerSheets),
    engineerSheets: pack_(engineerSheets),
    productionSheets: pack_(productionSheets),
    ftzSheets: pack_(ftzSheets),
    routingRuleVersion: 'V5.15',
    plannerFilterVersion: 'V5.17',
    plannerWeekRuleVersion: 'V5.23',
    ftzHistoryRuleVersion: 'V5.28',
    productionHistoryRuleVersion: 'V5.32'
  });
}

function saveSettings(settings) {
  const previous = getSettings();

  const clean = {
    plannerUrl: String(settings.plannerUrl || '').trim(),
    plannerTab: String(settings.plannerTab || '').trim(),
    plannerAllTabs: !!settings.plannerAllTabs,
    engineerUrl: String(settings.engineerUrl || '').trim(),
    engineerTab: String(settings.engineerTab || '').trim(),
    engineerAllTabs: !!settings.engineerAllTabs,
    productionUrl: String(settings.productionUrl || '').trim(),
    productionTab: String(settings.productionTab || '').trim(),
    productionAllTabs: !!settings.productionAllTabs,
    ftzUrl: String(settings.ftzUrl || '').trim(),
    ftzTab: String(settings.ftzTab || '').trim(),
    ftzAllTabs: !!settings.ftzAllTabs,
    mailTasksUrl: String(settings.mailTasksUrl || '').trim(),
    mailTasksTab: String(settings.mailTasksTab || MAIL_TASKS_DEFAULT_TAB).trim() || MAIL_TASKS_DEFAULT_TAB,
    chatMessagesUrl: String(settings.chatMessagesUrl || '').trim(),
    chatMessagesTab: String(settings.chatMessagesTab || CHAT_MESSAGES_DEFAULT_TAB).trim() || CHAT_MESSAGES_DEFAULT_TAB,
    componentAllocationUrl: String(settings.componentAllocationUrl || '').trim(),
    componentTemplateTab: String(settings.componentTemplateTab || 'TEMPLATE').trim() || 'TEMPLATE',
    refreshMinutes: Math.max(1, Math.min(60, Number(settings.refreshMinutes || 5)))
  };

  if (!clean.plannerUrl || !clean.engineerUrl) {
    throw new Error('Planner 和 Engineer Google Sheet link 都必须填写。');
  }

  const scriptProps = PropertiesService.getScriptProperties();
  const plannerChanged = !samePlannerHistorySource_(previous, clean);
  const otherSourcesChanged = !sameOtherHistorySources_(previous, clean);

  // V5.5:
  // If ONLY Masterplan changes, preserve the old snapshot long enough to
  // compare old/new Masterplan by S/N. High-confidence field changes are
  // recovered into Change History before the new sheet becomes the baseline.
  if (plannerChanged && !otherSourcesChanged) {
    scriptProps.setProperty(HISTORY_PLANNER_MIGRATION_KEY, '1');
    scriptProps.deleteProperty(HISTORY_FORCE_REBASE_KEY);
  } else if (plannerChanged || otherSourcesChanged) {
    // Engineer / Production source changes remain a safe rebase.
    scriptProps.setProperty(HISTORY_FORCE_REBASE_KEY, '1');
    scriptProps.deleteProperty(HISTORY_PLANNER_MIGRATION_KEY);
  }

  PropertiesService.getUserProperties()
    .setProperty(SETTINGS_KEY, JSON.stringify(clean));

  return clean;
}

function getDashboardData() {
  const settings = getSettings();
  if (!settings.plannerUrl || !settings.engineerUrl) {
    return {
      configured: false,
      settings: settings,
      rows: [],
      warnings: [],
      lastSync: formatDateTime_(new Date())
    };
  }

  const plannerSheets = openSheets_(settings.plannerUrl, settings.plannerTab, settings.plannerAllTabs);
  const engineerSheets = openSheets_(settings.engineerUrl, settings.engineerTab, settings.engineerAllTabs);

  const plannerParsed = parsePlannerSheets_(plannerSheets);
  const engineerParsed = parseEngineerSheets_(engineerSheets);

  let ftzSheets = [];
  let ftzParsed = { bySn: {}, warnings: [] };
  if (settings.ftzUrl) {
    ftzSheets = openSheets_(settings.ftzUrl, settings.ftzTab, settings.ftzAllTabs);
    ftzParsed = parseFtzSheets_(ftzSheets);
  }

  let productionSheets = [];
  let productionParsed = { rows: [], warnings: [] };
  if (settings.productionUrl) {
    productionSheets = openSheets_(settings.productionUrl, settings.productionTab, settings.productionAllTabs);
    productionParsed = parseProductionSheets_(
      productionSheets,
      plannerParsed.rows,
      engineerParsed.bySn,
      ftzParsed.bySn,
      !!settings.ftzUrl
    );
  }

  const rows = [];
  const warnings = []
    .concat(plannerParsed.warnings || [])
    .concat(engineerParsed.warnings || [])
    .concat(ftzParsed.warnings || [])
    .concat(productionParsed.warnings || []);

  plannerParsed.rows.forEach(function(p) {
    const key = normalizeSn_(p.sn);
    const e = engineerParsed.bySn[key] || null;
    const f = ftzParsed.bySn[key] || null;

    let dateMismatch = false;
    if (e && p.comDateKey && e.committedDateKey) {
      dateMismatch = p.comDateKey !== e.committedDateKey;
    }

    const engineerExempt = isEngineerExemptSn_(p.sn);

    // V7.21 Routing Workbench: orders on the same Engineer row are one Routing design group.
    const routingGroupSns = e
      ? expandSnRange_(e.snSource || p.sn).map(function(x){ return normalizeSnDisplay_(x); })
      : [normalizeSnDisplay_(p.sn)];
    const routingGroupNorm = routingGroupSns
      .map(function(x){ return normalizeSn_(x); })
      .filter(Boolean)
      .sort();
    const routingGroupKey = e && routingGroupNorm.length
      ? ('ENG|' + routingGroupNorm.join(','))
      : ('SN|' + normalizeSn_(p.sn));

    // Engineer Sheet is always the primary source.
    // Only when there is NO Engineer match, a FTZ Routing FINAL DONE
    // may backfill both Mechanical and Electrical as FINAL DONE.
    const ftzRoutingFinalFallback =
      !engineerExempt &&
      !e &&
      !!(f && f.routing && f.routing.stage === 'FINAL DONE');

    const effectiveEngineerMatched = !!e || ftzRoutingFinalFallback;

    rows.push({
      seq: p.seq,
      sapSo: p.sapSo,
      sn: p.sn,
      model: p.model,
      customer: p.customer,
      endUser: p.endUser,
      comDate: p.comDate,
      comDateKey: p.comDateKey,
      plannerWeek: p.plannerWeek,
      plannerYear: p.plannerYear,
      weekKey: p.weekKey,
      plannerRow: p.sourceRow,

      engineerMatched: effectiveEngineerMatched,
      engineerExempt: engineerExempt,
      engineerFallbackFromFtz: ftzRoutingFinalFallback,
      engineerFallbackSource: ftzRoutingFinalFallback ? 'FTZ ROUTING FINAL DONE' : '',
      engineerSnSource: e ? e.snSource : '',
      engineerGroupSns: routingGroupSns,
      engineerGroupKey: routingGroupKey,
      engineerCommittedDate: e ? e.committedDate : '',
      engineerCommittedDateKey: e ? e.committedDateKey : '',
      dateMismatch: dateMismatch,

      mechEngineer: e ? e.mechEngineer : '',
      elecEngineer: e ? e.elecEngineer : '',
      mechanical: e
        ? e.mechanical
        : (ftzRoutingFinalFallback ? emptyStatus_('FINAL DONE') : emptyStatus_('NO ENGINEER MATCH')),
      electrical: e
        ? e.electrical
        : (ftzRoutingFinalFallback ? emptyStatus_('FINAL DONE') : emptyStatus_('NO ENGINEER MATCH')),

      ftzConfigured: !!settings.ftzUrl,
      ftzMatched: !!f,
      routing: engineerExempt
        ? emptyRoutingStatus_('-')
        : (f
          ? f.routing
          : emptyRoutingStatus_(settings.ftzUrl ? 'NOT STARTED' : '-')),
      routingRemarks: f ? f.remarks : '',
      routingSourceTab: f ? f.sourceTab : ''
    });
  });

  const matched = rows.filter(r => r.engineerMatched).length;
  const engineerExempt = rows.filter(r => r.engineerExempt).length;
  const ready = rows.filter(r =>
    r.engineerMatched &&
    !r.engineerExempt &&
    r.mechanical.stage === 'FINAL DONE' &&
    r.electrical.stage === 'FINAL DONE'
  ).length;
  const overdue = rows.filter(r =>
    (r.mechanical && r.mechanical.overdue) ||
    (r.electrical && r.electrical.overdue)
  ).length;

  // Persist reference-sheet changes outside the browser.
  // This does NOT modify Planner / Engineer / Production / FTZ source sheets.
  const historySourceSignature = buildHistorySourceSignature_(
    settings,
    plannerSheets,
    engineerSheets,
    productionSheets,
    ftzSheets
  );

  const historyResult = syncPersistentHistory_(
    rows,
    productionParsed.rows || [],
    historySourceSignature
  );

  return {
    configured: true,
    settings: settings,
    plannerSheetName: plannerSheets.map(s => s.getName()).join(', '),
    engineerSheetName: engineerSheets.map(s => s.getName()).join(', '),
    productionSheetName: productionSheets.length ? productionSheets.map(s => s.getName()).join(', ') : '',
    ftzSheetName: ftzSheets.length ? ftzSheets.map(s => s.getName()).join(', ') : '',
    productionRows: productionParsed.rows || [],
    changeHistory: historyResult.items || [],
    changeHistoryTotal: historyResult.total || 0,
    historyRebased: !!historyResult.rebased,
    historyRebaseReason: historyResult.rebaseReason || '',
    historyReset: !!historyResult.historyReset,
    migrationReport: historyResult.migrationReport || null,
    lastSync: formatDateTime_(new Date()),
    summary: {
      total: rows.length,
      matched: matched,
      ready: ready,
      overdue: overdue,
      unmatched: rows.filter(r => !r.engineerMatched && !r.engineerExempt).length
    },
    rows: rows,
    warnings: warnings
  };
}

function ensureMailDoneArchive_(ss) {
  let sh = ss.getSheetByName(WORKBENCH_MAIL_DONE_SHEET);
  if (!sh) sh = ss.insertSheet(WORKBENCH_MAIL_DONE_SHEET);

  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, 15).setValues([[
      'Task Key',

      'Summary',
      'Priority',
      'Due Date',
      'S/N',

      'ECO',
      'Suggested Action',
      'Email Link'
    ]]);
    sh.setFrozenRows(1);
  }
  return sh;
}

function mailTaskStableKey_(task) {
  task = task || {};
  const raw = [
    String(task.received || '').trim(),
    String(task.from || '').trim().toLowerCase(),
    String(task.subject || '').trim().toLowerCase(),
    String(task.taskTitle || '').trim().toLowerCase()
  ].join('\n');

  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    raw,
    Utilities.Charset.UTF_8
  );
  return bytes.map(function(b) {
    const n = b < 0 ? b + 256 : b;
    return ('0' + n.toString(16)).slice(-2);
  }).join('');
}

function syncDoneTasksToArchive_(tasks) {
  const done = (tasks || []).filter(function(t) {
    return normalizeMailTaskStatus_(t.status) === 'DONE' && !t.archivedOnly;
  });
  if (!done.length) return;

  const ss = getOrCreateWorkbenchStore_();
  const sh = ensureMailDoneArchive_(ss);
  const existing = {};
  const lastRow = sh.getLastRow();
  if (lastRow >= 2) {
    sh.getRange(2, 1, lastRow - 1, 1).getDisplayValues().forEach(function(r) {
      const key = String((r || [])[0] || '').trim();
      if (key) existing[key] = true;
    });
  }

  const now = formatDateTime_(new Date());
  const rows = [];
  done.forEach(function(t) {
    const key = mailTaskStableKey_(t);
    if (existing[key]) return;
    existing[key] = true;
    rows.push([
      key,
      now,
      t.received || '',
      t.from || '',
      t.subject || '',
      t.taskTitle || '',
      t.category || 'Other',
      t.summary || '',
      t.priority || 'Normal',
      t.dueDate || '',
      t.sn || '',
      t.sapSo || '',
      t.eco || '',
      t.suggestedAction || '',
      t.emailLink || ''
    ]);
  });

  if (rows.length) {
    sh.getRange(sh.getLastRow() + 1, 1, rows.length, 15).setValues(rows);
    SpreadsheetApp.flush();
  }
}

function readDoneTaskArchive_() {
  const ss = getOrCreateWorkbenchStore_();
  const sh = ensureMailDoneArchive_(ss);
  const lastRow = sh.getLastRow();
  if (lastRow < 2) return [];

  const values = sh.getRange(2, 1, lastRow - 1, 15).getDisplayValues();
  return values.map(function(r, i) {
    const key = String(r[0] || '').trim();
    return {
      rowNumber: -(i + 2),
      archiveKey: key,
      archivedOnly: true,
      completedAt: String(r[1] || '').trim(),
      received: String(r[2] || '').trim(),
      from: String(r[3] || '').trim(),
      subject: String(r[4] || '').trim(),
      taskTitle: String(r[5] || '').trim() || String(r[4] || '').trim() || 'Untitled task',
      category: String(r[6] || '').trim() || 'Other',
      summary: String(r[7] || '').trim(),
      priority: String(r[8] || '').trim() || 'Normal',
      dueDate: String(r[9] || '').trim(),
      sn: String(r[10] || '').trim(),
      sapSo: String(r[11] || '').trim(),
      eco: String(r[12] || '').trim(),
      suggestedAction: String(r[13] || '').trim(),
      status: 'DONE',
      emailLink: String(r[14] || '').trim()
    };
  }).filter(function(t) {
    return !!(t.archiveKey || t.taskTitle || t.subject);
  });
}

// V7.7: lightweight Mail Tasks change detector.
// This polls only the tail of the task sheet, then performs a full inbox refresh only when data changed.
function mailTaskInboxPulseForSheet_(sh) {
  const lastRow = Math.max(0, Number(sh.getLastRow() || 0));
  const lastCol = Math.min(20, Math.max(1, Number(sh.getLastColumn() || 1)));
  let tail = '';
  if (lastRow >= 2) {
    const startRow = Math.max(2, lastRow - 1); // last two data rows are enough to detect new Studio appends
    const values = sh.getRange(startRow, 1, lastRow - startRow + 1, lastCol).getDisplayValues();
    tail = values.map(function(row) { return row.join('\u001f'); }).join('\u001e');
  }
  const raw = [lastRow, lastCol, tail].join('|');
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, raw, Utilities.Charset.UTF_8);
  return {
    pulse: Utilities.base64EncodeWebSafe(bytes).replace(/=+$/g, ''),
    lastRow: lastRow,
    checkedAt: formatDateTime_(new Date())
  };
}

function getMailTaskInboxPulse() {
  const settings = getSettings();
  const url = String(settings.mailTasksUrl || '').trim();
  const tab = String(settings.mailTasksTab || MAIL_TASKS_DEFAULT_TAB).trim() || MAIL_TASKS_DEFAULT_TAB;
  if (!url) return { configured: false, pulse: '', lastRow: 0, checkedAt: formatDateTime_(new Date()) };
  const sh = openSheets_(url, tab, false)[0];
  const p = mailTaskInboxPulseForSheet_(sh);
  p.configured = true;
  return p;
}

function getMailTaskInbox(forceRefresh) {
  const settings = getSettings();
  const url = String(settings.mailTasksUrl || '').trim();
  const tab = String(settings.mailTasksTab || MAIL_TASKS_DEFAULT_TAB).trim() || MAIL_TASKS_DEFAULT_TAB;
  const cache = CacheService.getUserCache();
  const cacheKey = 'mail_task_inbox_v75';

  if (!forceRefresh) {
    const cached = cache.get(cacheKey);
    if (cached) {
      try { return JSON.parse(cached); } catch (e) {}
    }
  }

  if (!url) {
    return {
      configured: false,
      tasks: [],
      sheetName: '',
      spreadsheetName: '',
      loadedAt: formatDateTime_(new Date()),
      message: 'Mail Tasks Google Sheet is not configured.'
    };
  }

  const sh = openSheets_(url, tab, false)[0];
  const pulseInfo = mailTaskInboxPulseForSheet_(sh);
  const data = sh.getDataRange().getDisplayValues();
  if (!data.length) {
    const emptyResult = {
      configured: true,
      tasks: [],
      sheetName: sh.getName(),
      spreadsheetName: sh.getParent().getName(),
      loadedAt: formatDateTime_(new Date()),
      pulse: pulseInfo.pulse,
      message: 'Mail Tasks sheet is empty.'
    };
    try { cache.put(cacheKey, JSON.stringify(emptyResult), 45); } catch (e) {}
    return emptyResult;
  }

  const cols = mailTaskColumns_(data[0]);
  if (cols.taskTitle == null || cols.status == null) {
    throw new Error('Mail Tasks sheet must contain at least "Task Title" and "Status" columns in row 1.');
  }

  const tasks = [];
  for (let r = 1; r < data.length; r++) {
    const row = data[r] || [];
    const taskTitle = mailTaskCell_(row, cols.taskTitle);
    const subject = mailTaskCell_(row, cols.subject);
    const summary = mailTaskCell_(row, cols.summary);
    if (!taskTitle && !subject && !summary) continue;

    const status = normalizeMailTaskStatus_(mailTaskCell_(row, cols.status));
    tasks.push({
      rowNumber: r + 1,
      received: mailTaskCell_(row, cols.received),
      from: mailTaskCell_(row, cols.from),
      subject: subject,
      taskTitle: taskTitle || subject || 'Untitled task',
      category: mailTaskCell_(row, cols.category) || 'Other',
      summary: summary,
      priority: mailTaskCell_(row, cols.priority) || 'Normal',
      dueDate: mailTaskCell_(row, cols.dueDate),
      sn: mailTaskCell_(row, cols.sn),
      sapSo: mailTaskCell_(row, cols.sapSo),
      eco: mailTaskCell_(row, cols.eco),
      suggestedAction: mailTaskCell_(row, cols.suggestedAction),
      status: status,
      emailLink: mailTaskCell_(row, cols.emailLink)
    });
  }

  // V6.7+: completed tasks become a permanent Workbench archive.
  syncDoneTasksToArchive_(tasks);

  const liveKeys = {};
  tasks.forEach(function(t) {
    liveKeys[mailTaskStableKey_(t)] = true;
  });

  readDoneTaskArchive_().forEach(function(t) {
    if (!liveKeys[t.archiveKey]) tasks.push(t);
  });

  const result = {
    configured: true,
    tasks: tasks,
    sheetName: sh.getName(),
    spreadsheetName: sh.getParent().getName(),
    loadedAt: formatDateTime_(new Date()),
    pulse: pulseInfo.pulse,
    message: ''
  };

  // Short per-user cache: fast tab switching without hiding new Studio tasks for long.
  try { cache.put(cacheKey, JSON.stringify(result), 45); } catch (e) {}
  return result;
}

function setMailTaskStatus(taskRef, status) {
  const settings = getSettings();
  const url = String(settings.mailTasksUrl || '').trim();
  const tab = String(settings.mailTasksTab || MAIL_TASKS_DEFAULT_TAB).trim() || MAIL_TASKS_DEFAULT_TAB;
  if (!url) throw new Error('Mail Tasks Google Sheet is not configured.');

  status = normalizeMailTaskStatus_(status);
  if (['NEW', 'IN PROGRESS', 'DONE', 'IGNORED'].indexOf(status) === -1) {
    throw new Error('Unsupported task status: ' + status);
  }

  taskRef = taskRef || {};
  const sh = openSheets_(url, tab, false)[0];
  const data = sh.getDataRange().getDisplayValues();
  if (!data.length) throw new Error('Mail Tasks sheet is empty.');

  const cols = mailTaskColumns_(data[0]);
  if (cols.status == null) throw new Error('Mail Tasks sheet is missing the Status column.');

  let targetRow = Number(taskRef.rowNumber || 0);
  if (!mailTaskRefMatchesRow_(data, cols, targetRow, taskRef)) {
    targetRow = 0;
    for (let r = 1; r < data.length; r++) {
      if (mailTaskRefMatchesRow_(data, cols, r + 1, taskRef)) {
        targetRow = r + 1;
        break;
      }
    }
  }

  if (!targetRow) {
    throw new Error('Task row could not be found. Refresh Workbench and try again.');
  }

  sh.getRange(targetRow, cols.status + 1).setValue(status);
  SpreadsheetApp.flush();
  try { CacheService.getUserCache().remove('mail_task_inbox_v75'); } catch (e) {}

  if (status === 'DONE') {
    const row = data[targetRow - 1] || [];
    const doneTask = {
      rowNumber: targetRow,
      received: mailTaskCell_(row, cols.received),
      from: mailTaskCell_(row, cols.from),
      subject: mailTaskCell_(row, cols.subject),
      taskTitle: mailTaskCell_(row, cols.taskTitle) || mailTaskCell_(row, cols.subject) || 'Untitled task',
      category: mailTaskCell_(row, cols.category) || 'Other',
      summary: mailTaskCell_(row, cols.summary),
      priority: mailTaskCell_(row, cols.priority) || 'Normal',
      dueDate: mailTaskCell_(row, cols.dueDate),
      sn: mailTaskCell_(row, cols.sn),
      sapSo: mailTaskCell_(row, cols.sapSo),
      eco: mailTaskCell_(row, cols.eco),
      suggestedAction: mailTaskCell_(row, cols.suggestedAction),
      status: 'DONE',
      emailLink: mailTaskCell_(row, cols.emailLink)
    };
    syncDoneTasksToArchive_([doneTask]);
  }

  return getMailTaskInbox(true);
}

function mailTaskRefMatchesRow_(data, cols, rowNumber, ref) {
  if (!rowNumber || rowNumber < 2 || rowNumber > data.length) return false;
  const row = data[rowNumber - 1] || [];
  const subject = mailTaskCell_(row, cols.subject);
  const received = mailTaskCell_(row, cols.received);
  const taskTitle = mailTaskCell_(row, cols.taskTitle);

  const expectedSubject = String(ref.subject || '').trim();
  const expectedReceived = String(ref.received || '').trim();
  const expectedTaskTitle = String(ref.taskTitle || '').trim();

  if (expectedSubject && subject !== expectedSubject) return false;
  if (expectedReceived && received !== expectedReceived) return false;
  if (expectedTaskTitle && taskTitle !== expectedTaskTitle) return false;
  return !!(subject || received || taskTitle);
}

function mailTaskColumns_(headerRow) {
  const aliases = {
    received: ['received', 'received date', 'received time', 'date received'],
    from: ['from', 'sender'],
    subject: ['subject', 'email subject'],
    taskTitle: ['task title', 'task'],
    category: ['category', 'task category', 'type'],
    summary: ['summary', 'task summary'],
    priority: ['priority'],
    dueDate: ['due date', 'deadline'],
    sn: ['s/n', 'sn', 'serial number'],
    sapSo: ['sap so', 'sales order', 'sap sales order'],
    eco: ['eco', 'eco number'],
    suggestedAction: ['suggested action', 'next action'],
    status: ['status', 'task status'],
    emailLink: ['email link', 'gmail link', 'message link']
  };

  const normalized = (headerRow || []).map(normalizeHeader_);
  const out = {};
  Object.keys(aliases).forEach(function(key) {
    const choices = aliases[key].map(normalizeHeader_);
    let idx = null;
    for (let i = 0; i < normalized.length; i++) {
      if (choices.indexOf(normalized[i]) !== -1) {
        idx = i;
        break;
      }
    }
    out[key] = idx;
  });
  return out;
}

function mailTaskCell_(row, idx) {
  return idx == null ? '' : String((row || [])[idx] == null ? '' : row[idx]).trim();
}

function normalizeMailTaskStatus_(value) {
  const s = String(value || '').trim().toUpperCase().replace(/[_-]+/g, ' ').replace(/\s+/g, ' ');
  if (!s) return 'NEW';
  if (s === 'NEW') return 'NEW';
  if (s === 'IN PROGRESS' || s === 'STARTED' || s === 'WORKING') return 'IN PROGRESS';
  if (s === 'DONE' || s === 'COMPLETED' || s === 'COMPLETE') return 'DONE';
  if (s === 'IGNORED' || s === 'IGNORE') return 'IGNORED';
  return s;
}

// V7.4 · Reply in original Gmail conversation with optional user-selected attachments.
function replyToMailTaskWithAttachments(taskRef, replyBody, replyAll, attachmentPayload) {
  taskRef = taskRef || {};
  replyBody = String(replyBody || '').trim();
  if (!replyBody) throw new Error('Reply message is empty.');

  let msg;
  try {
    msg = findMailTaskMessage_(taskRef);
  } catch (e) {
    throw new Error(
      'Gmail access is not authorized. Run testGmailConnection() once in Apps Script and allow Gmail access. ' +
      (e && e.message ? e.message : '')
    );
  }
  if (!msg) throw new Error('Could not safely match this task to the original Gmail message.');

  const payloads = Array.isArray(attachmentPayload) ? attachmentPayload : [];
  const blobs = [];
  let totalBytes = 0;
  payloads.forEach(function(item, index) {
    item = item || {};
    const base64 = String(item.base64 || '').trim();
    if (!base64) return;
    let bytes;
    try { bytes = Utilities.base64Decode(base64); }
    catch (e) { throw new Error('Attachment ' + (index + 1) + ' could not be decoded.'); }
    totalBytes += bytes.length;
    if (totalBytes > 8 * 1024 * 1024) throw new Error('Attachments exceed the 8 MB Workbench limit.');
    const name = String(item.name || ('attachment-' + (index + 1))).replace(/[\\/:*?"<>|]/g, '_');
    const mimeType = String(item.mimeType || 'application/octet-stream');
    blobs.push(Utilities.newBlob(bytes, mimeType, name));
  });

  const options = blobs.length ? {attachments: blobs} : {};
  if (replyAll) msg.replyAll(replyBody, options);
  else msg.reply(replyBody, options);

  const thread = msg.getThread();
  return {
    ok: true,
    message: replyAll ? 'Reply all sent.' : 'Reply sent.',
    subject: String(msg.getSubject() || ''),
    to: extractEmailAddress_(msg.getFrom()) || String(msg.getFrom() || ''),
    attachmentCount: blobs.length,
    permalink: thread ? thread.getPermalink() : ''
  };
}


// ============================================================
// V7.7 · Mail performance helpers
// Cache only per-user Gmail identifiers / safe viewer payloads.
// ============================================================
function mailTaskCacheHash_(taskRef) {
  taskRef = taskRef || {};
  const raw = [
    String(taskRef.received || '').trim(),
    String(taskRef.from || '').trim().toLowerCase(),
    normalizeMailSubjectForMatch_(taskRef.subject || ''),
    String(taskRef.taskTitle || '').trim().toLowerCase()
  ].join('|');
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, raw, Utilities.Charset.UTF_8);
  return Utilities.base64EncodeWebSafe(bytes).replace(/=+$/g, '');
}

function getCachedMailTaskMessage_(taskRef) {
  const cache = CacheService.getUserCache();
  const key = 'mail_msgid_v75_' + mailTaskCacheHash_(taskRef);
  const id = cache.get(key);
  if (!id) return null;
  try {
    const msg = GmailApp.getMessageById(id);
    if (!msg) return null;
    const wantedSubject = normalizeMailSubjectForMatch_(taskRef.subject || '');
    const gotSubject = normalizeMailSubjectForMatch_(msg.getSubject() || '');
    if (wantedSubject && gotSubject !== wantedSubject) return null;
    const wantedFrom = extractEmailAddress_(taskRef.from || '').toLowerCase();
    const gotFrom = extractEmailAddress_(msg.getFrom() || '').toLowerCase();
    if (wantedFrom && gotFrom && wantedFrom !== gotFrom) return null;
    return msg;
  } catch (e) {
    try { cache.remove(key); } catch (ignore) {}
    return null;
  }
}

function putCachedMailTaskMessage_(taskRef, msg) {
  if (!msg) return;
  try {
    CacheService.getUserCache().put(
      'mail_msgid_v75_' + mailTaskCacheHash_(taskRef),
      String(msg.getId() || ''),
      21600
    );
  } catch (e) {}
}

function getMailTaskOriginalEmail(taskRef, forceRefresh) {
  taskRef = taskRef || {};
  const cache = CacheService.getUserCache();
  const viewerKey = 'mail_original_v814_' + mailTaskCacheHash_(taskRef);

  if (!forceRefresh) {
    const cached = cache.get(viewerKey);
    if (cached) {
      try { return JSON.parse(cached); } catch (e) {}
    }
  }

  let msg;
  try {
    msg = findMailTaskMessage_(taskRef);
  } catch (e) {
    throw new Error(
      'Gmail access is not authorized. Run testGmailConnection() once in Apps Script and allow Gmail access. ' +
      (e && e.message ? e.message : '')
    );
  }

  if (!msg) {
    throw new Error('Could not safely match this task to the original Gmail message.');
  }

  const thread = msg.getThread();
  let attachments = [];
  try {
    attachments = (msg.getAttachments({includeInlineImages:false, includeAttachments:true}) || []).map(function(a) {
      return {
        name: String(a.getName() || 'Attachment'),
        size: Number(a.getBytes ? a.getBytes().length : 0) || 0
      };
    });
  } catch (e) {
    attachments = [];
  }

  let body = '';
  try { body = String(msg.getPlainBody() || ''); } catch (e) { body = ''; }
  if (body.length > 50000) body = body.slice(0, 50000) + '\n\n[Message truncated in Workbench]';

  const imageParts = routingCollectMessageImages_(msg);
  // Keep the legacy payload so an older deployed Index.html still shows images.
  // V8.1.10 Index.html uses htmlBody and therefore does not render this gallery.
  const inlineImages = routingImagePayload_(imageParts);
  const htmlBody = routingSafeMessageHtml_(msg, imageParts, inlineImages);
  const mailLooksRouting = /\b(?:routing|prelist|final\s+drawing|planning\s*\/\s*purchasing)\b/i.test([
    msg.getSubject ? msg.getSubject() : '',
    taskRef.category || '',
    taskRef.taskTitle || ''
  ].join(' '));
  const ocrInfo = mailLooksRouting
    ? routingOcrMessageImages_(msg, imageParts)
    : {text:'', error:''};

  const result = {
    ok: true,
    id: String(msg.getId() || ''),
    threadId: thread ? String(thread.getId() || '') : '',
    permalink: thread ? String(thread.getPermalink() || '') : '',
    subject: String(msg.getSubject() || ''),
    from: String(msg.getFrom() || ''),
    to: String(msg.getTo() || ''),
    cc: String(msg.getCc() || ''),
    date: msg.getDate() ? formatDateTime_(msg.getDate()) : '',
    body: body,
    htmlBody: htmlBody,
    attachments: attachments,
    inlineImages: inlineImages,
    ocrIds: extractRoutingBrIdsLoose_(ocrInfo.text),
    ocrEvidence: ocrInfo.evidence || [],
    ocrError: ocrInfo.error || ''
  };

  try { cache.put(viewerKey, JSON.stringify(result), 600); } catch (e) {}
  return result;
}

function normalizeMailSenderNameForMatch_(value) {
  return String(value || '')
    .replace(/<[^>]*>/g, ' ')
    .replace(/[^a-z0-9]+/gi, ' ')
    .toLowerCase()
    .replace(/\s+/g, ' ')
    .trim();
}

function mailTaskDistinctiveTokens_(taskRef) {
  taskRef = taskRef || {};
  const joined = [
    taskRef.subject || '', taskRef.taskTitle || '', taskRef.sn || '',
    taskRef.sapSo || '', taskRef.eco || ''
  ].join(' ');

  const ids = [];
  const seenId = {};
  const idRx = /\b(?:ECO[A-Z0-9-]*\d[A-Z0-9-]*|BR\d{5,}|[A-Z]{2,6}\d{5,}|\d{5,}(?:-\d{1,5})?)\b/gi;
  let m;
  while ((m = idRx.exec(joined)) !== null) {
    const v = String(m[0] || '').toUpperCase();
    if (!seenId[v]) { seenId[v] = true; ids.push(v); }
    if (ids.length >= 8) break;
  }

  const stop = {
    'with':1,'from':1,'this':1,'that':1,'have':1,'your':1,'please':1,'email':1,
    'task':1,'routing':1,'review':1,'create':1,'update':1,'final':1,'release':1,
    'production':1,'order':1,'design':1,'drawing':1,'drawings':1,'reply':1,'mail':1
  };
  const words = normalizeMailSubjectForMatch_(String(taskRef.subject || '') + ' ' + String(taskRef.taskTitle || ''))
    .replace(/[^a-z0-9]+/g, ' ')
    .split(/\s+/)
    .filter(function(x) { return x && x.length >= 4 && !stop[x] && !/^\d+$/.test(x); });
  const seenWord = {};
  const uniqWords = [];
  words.forEach(function(w) {
    if (!seenWord[w]) { seenWord[w] = true; uniqWords.push(w); }
  });
  return { ids: ids, words: uniqWords.slice(0, 8) };
}

function mailTaskReceivedMs_(value) {
  let s = String(value || '').trim();
  if (!s) return 0;
  // Workspace Studio often formats dates as "September 14, 2026 at 10:12 AM EDT".
  s = s.replace(/\s+at\s+/i, ' ');
  let d = new Date(s);
  if (!isNaN(d.getTime())) return d.getTime();
  // Fallback: remove an unrecognised trailing timezone abbreviation.
  d = new Date(s.replace(/\s+[A-Z]{2,5}$/i, ''));
  return isNaN(d.getTime()) ? 0 : d.getTime();
}

function scoreMailTaskMessage_(msg, taskRef, tokenInfo) {
  taskRef = taskRef || {};
  tokenInfo = tokenInfo || mailTaskDistinctiveTokens_(taskRef);

  const wantedSubjectRaw = String(taskRef.subject || '').trim();
  const wantedSubject = normalizeMailSubjectForMatch_(wantedSubjectRaw);
  const gotSubjectRaw = String(msg.getSubject() || '').trim();
  const gotSubject = normalizeMailSubjectForMatch_(gotSubjectRaw);

  const wantedFromEmail = extractEmailAddress_(taskRef.from || '').toLowerCase();
  const gotFromRaw = String(msg.getFrom() || '');
  const gotFromEmail = extractEmailAddress_(gotFromRaw).toLowerCase();
  const wantedFromName = normalizeMailSenderNameForMatch_(taskRef.from || '');
  const gotFromName = normalizeMailSenderNameForMatch_(gotFromRaw);

  // If Studio captured a real sender address, never cross-match a different sender.
  if (wantedFromEmail && gotFromEmail && wantedFromEmail !== gotFromEmail) return null;

  let score = 0;
  let exactSubject = false;
  let senderMatched = false;
  let dateDiff = Number.POSITIVE_INFINITY;

  if (wantedSubject && gotSubject) {
    if (wantedSubject === gotSubject) { score += 120; exactSubject = true; }
    else if (gotSubject.indexOf(wantedSubject) !== -1 || wantedSubject.indexOf(gotSubject) !== -1) score += 55;
  }

  if (wantedFromEmail && gotFromEmail && wantedFromEmail === gotFromEmail) {
    score += 35; senderMatched = true;
  } else if (wantedFromName && gotFromName &&
      (wantedFromName === gotFromName || gotFromName.indexOf(wantedFromName) !== -1 || wantedFromName.indexOf(gotFromName) !== -1)) {
    score += 25; senderMatched = true;
  }

  const targetMs = mailTaskReceivedMs_(taskRef.received || '');
  const msgMs = msg.getDate() ? msg.getDate().getTime() : 0;
  if (targetMs && msgMs) {
    dateDiff = Math.abs(targetMs - msgMs);
    if (dateDiff <= 2 * 60 * 1000) score += 80;
    else if (dateDiff <= 60 * 60 * 1000) score += 55;
    else if (dateDiff <= 24 * 60 * 60 * 1000) score += 35;
    else if (dateDiff <= 3 * 24 * 60 * 60 * 1000) score += 10;
  }

  let body = '';
  let haystack = gotSubject.toUpperCase();
  if ((tokenInfo.ids || []).length || (tokenInfo.words || []).length) {
    try { body = String(msg.getPlainBody() || '').slice(0, 30000); } catch (e) { body = ''; }
    haystack += '\n' + body.toUpperCase();
  }

  let idHits = 0;
  (tokenInfo.ids || []).forEach(function(id) {
    if (haystack.indexOf(String(id).toUpperCase()) !== -1) idHits++;
  });
  if (idHits) score += Math.min(160, idHits * 80);

  let wordHits = 0;
  const lowHay = haystack.toLowerCase();
  (tokenInfo.words || []).forEach(function(w) {
    if (lowHay.indexOf(w) !== -1) wordHits++;
  });
  if (wordHits) score += Math.min(45, wordHits * 9);

  // Safety gate. Viewing/replying must have at least one strong identity signal.
  const strong = exactSubject || idHits > 0 ||
    (senderMatched && dateDiff <= 24 * 60 * 60 * 1000) ||
    (senderMatched && wordHits >= 2);
  if (!strong) return null;

  return {
    msg: msg,
    score: score,
    exactSubject: exactSubject,
    idHits: idHits,
    senderMatched: senderMatched,
    dateDiff: dateDiff,
    dateMs: msgMs
  };
}

function gmailThreadFromTaskLink_(link) {
  const s = String(link || '').trim();
  if (!s) return null;
  // Gmail permalinks commonly end with a thread id after #inbox/, #all/, etc.
  const m = s.match(/(?:#|\/)(?:inbox\/|all\/|sent\/|search\/[^/]+\/)?([0-9a-f]{12,})\b/i);
  if (!m) return null;
  try { return GmailApp.getThreadById(m[1]); } catch (e) { return null; }
}

function findMailTaskMessage_(taskRef) {
  taskRef = taskRef || {};
  const cachedMessage = getCachedMailTaskMessage_(taskRef);
  if (cachedMessage) return cachedMessage;

  const tokenInfo = mailTaskDistinctiveTokens_(taskRef);
  const targetSubject = normalizeMailSubjectForMatch_(taskRef.subject || '');
  const fromEmail = extractEmailAddress_(taskRef.from || '');
  const candidates = [];
  const seenMsg = {};

  function consider(msg) {
    if (!msg) return;
    const id = String(msg.getId() || '');
    if (id && seenMsg[id]) return;
    if (id) seenMsg[id] = true;
    const scored = scoreMailTaskMessage_(msg, taskRef, tokenInfo);
    if (scored) candidates.push(scored);
  }

  // Best path: if Studio/Sheet already contains a Gmail link, inspect only that thread.
  const linkedThread = gmailThreadFromTaskLink_(taskRef.emailLink || '');
  if (linkedThread) {
    try { linkedThread.getMessages().forEach(consider); } catch (e) {}
  }

  const queries = [];
  function addQuery(q) { if (q && queries.indexOf(q) === -1) queries.push(q); }

  // Strong business identifiers are much safer than requiring an exact subject.
  (tokenInfo.ids || []).slice(0, 3).forEach(function(id) {
    addQuery('newer_than:365d "' + id.replace(/"/g, '') + '"' + (fromEmail ? ' from:' + fromEmail : ''));
  });

  if (fromEmail) addQuery('newer_than:365d from:' + fromEmail);

  const subjectWords = targetSubject.replace(/[^a-z0-9]+/gi, ' ').trim().split(/\s+/)
    .filter(function(x) { return x.length >= 4; }).slice(0, 4);
  if (subjectWords.length) addQuery('newer_than:365d ' + subjectWords.map(function(x) { return '"' + x + '"'; }).join(' '));

  if ((tokenInfo.words || []).length) {
    addQuery('newer_than:365d ' + tokenInfo.words.slice(0, 3).map(function(x) { return '"' + x + '"'; }).join(' '));
  }

  // Last fallback is deliberately narrow in time; scoring still enforces a strong signal.
  addQuery('newer_than:30d');

  const threadMap = {};
  for (let q = 0; q < queries.length; q++) {
    const max = q === queries.length - 1 ? 50 : 100;
    const threads = GmailApp.search(queries[q], 0, max) || [];
    threads.forEach(function(t) { threadMap[t.getId()] = t; });
    if (Object.keys(threadMap).length >= 120) break;
  }

  Object.keys(threadMap).forEach(function(id) {
    try { threadMap[id].getMessages().forEach(consider); } catch (e) {}
  });

  candidates.sort(function(a, b) {
    if (b.score !== a.score) return b.score - a.score;
    return b.dateMs - a.dateMs;
  });

  if (!candidates.length) return null;
  const best = candidates[0];
  const second = candidates.length > 1 ? candidates[1] : null;

  // Avoid choosing between near-identical weak candidates. Exact subject or business ID
  // matches are allowed through because they are strong business identity signals.
  if (second && !best.exactSubject && best.idHits === 0 && (best.score - second.score) < 18) return null;
  if (best.score < 70) return null;

  putCachedMailTaskMessage_(taskRef, best.msg);
  return best.msg;
}

function normalizeMailSubjectForMatch_(value) {
  let s = String(value || '').trim();
  // Remove stacked prefixes such as "Fwd: Re: FW:" but keep the business
  // identity (ECO / SN / order wording) untouched.
  while (/^\s*(?:re|fw|fwd)\s*:\s*/i.test(s)) {
    s = s.replace(/^\s*(?:re|fw|fwd)\s*:\s*/i, '');
  }
  return s.toLowerCase().replace(/\s+/g, ' ').trim();
}

function extractEmailAddress_(value) {
  const s = String(value || '').trim();
  const angle = s.match(/<\s*([^<>\s]+@[^<>\s]+)\s*>/);
  if (angle) return angle[1].trim();
  const plain = s.match(/([A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,})/i);
  return plain ? plain[1].trim() : '';
}

function routingNewMessageText_(body) {
  const lines = String(body || '').replace(/\r/g, '').split('\n');
  const out = [];
  for (let i = 0; i < lines.length; i++) {
    const line = String(lines[i] || '');
    if (/^\s*>/.test(line)) break;
    if (/^\s*On\s.+wrote:\s*$/i.test(line)) break;
    if (/^\s*-{2,}\s*Original Message\s*-{2,}\s*$/i.test(line)) break;
    if (i > 1 && /^\s*From:\s+/i.test(line)) break;
    out.push(line);
  }
  return out.join('\n').trim();
}

function routingSenderDisplayName_(fromValue) {
  const s = String(fromValue || '').trim();
  const m = s.match(/^\s*"?([^"<]+?)"?\s*<[^>]+>/);
  if (m && m[1]) return m[1].trim();
  return s.replace(/<[^>]+>/g, '').trim() || s;
}

function routingSenderEmail_(fromValue) {
  const s = String(fromValue || '');
  const m = s.match(/([A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,})/i);
  return m ? m[1].toLowerCase() : '';
}

function routingCompanyDomain_() {

  try { email = String(Session.getActiveUser().getEmail() || ''); } catch (e) {}

function routingReplySignal_(body) {
  const text = routingNewMessageText_(body).replace(/\s+/g, ' ').trim();
  if (!text) return '';
  if (/\b(?:routing\s+)?(?:done|completed|complete)\b/i.test(text) ||
      /\b(?:done|completed)\s+(?:routing|prelist|final)\b/i.test(text) ||
      /\b(?:already|has\s+been)\s+(?:done|completed|updated|set)\b/i.test(text) ||
      /\bplease\s+proceed\b/i.test(text)) return 'DONE';
  if (/\b(?:checking|working\s+on|will\s+do|in\s+progress|processing)\b/i.test(text)) return 'IN PROGRESS';
  return '';
}

function routingReplyStateForThread_(messages, afterDate) {
  const domain = routingCompanyDomain_();
  const candidates = [];
  (messages || []).forEach(function(msg) {
    const d = msg.getDate ? msg.getDate() : null;
    if (!d || (afterDate && d.getTime() <= afterDate.getTime())) return;
    const from = String(msg.getFrom ? msg.getFrom() : '');
    const email = routingSenderEmail_(from);
    if (isRoutingHiddenSender_(from)) return;
    if (domain && email && email.split('@').pop() !== domain) return;
    let body = '';
    try { body = String(msg.getPlainBody() || ''); } catch (e) { body = ''; }
    const signal = routingReplySignal_(body);
    if (!signal) return;
    candidates.push({
      state: signal,
      by: routingSenderDisplayName_(from),
      email: email,
      at: d,
      atText: formatDateTime_(d)
    });
  });
  if (!candidates.length) return {state:'', by:'', at:'', email:''};
  candidates.sort(function(a,b){ return b.at.getTime() - a.at.getTime(); });
  const done = candidates.find(function(x){ return x.state === 'DONE'; });
  return done || candidates[0];
}

function routingImageContentType_(value, name) {
  let type = String(value || '').toLowerCase().split(';')[0].trim();
  if (type === 'image/jpg') type = 'image/jpeg';
  if (/^image\/(?:png|jpeg|gif|bmp|webp)$/i.test(type)) return type;
  const m = String(name || '').toLowerCase().match(/\.([a-z0-9]+)$/);
  const ext = m ? m[1] : '';
  return ({png:'image/png',jpg:'image/jpeg',jpeg:'image/jpeg',gif:'image/gif',bmp:'image/bmp',webp:'image/webp'})[ext] || '';
}

// V7.28: Outlook and Gmail do not always expose pasted screenshots in the same
// attachment collection. Collect inline parts, normal attachments, and finally
// raw MIME image parts. De-duplicate by content digest instead of filename.
function routingCollectMessageImages_(msg) {
  const out = [], seen = {};
  let totalBytes = 0;
  // Forwarded threads can contain several signatures before the actual BR list.
  // Keep all reasonably sized evidence images for the reader and OCR.
  const maxImages = 24, maxTotalBytes = 16000000;

  function addBytes_(bytes, contentType, name, source, contentId) {
    const type = routingImageContentType_(contentType, name);
    const size = Number(bytes && bytes.length || 0);
    if (!type || size < 1500 || size > 8000000 || totalBytes + size > maxTotalBytes || out.length >= maxImages) return;
    let sig = String(name || '') + '|' + type + '|' + size;
    try {
      sig = Utilities.base64EncodeWebSafe(
        Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, bytes)
      ).replace(/=+$/, '');
    } catch (e) {}
    if (Object.prototype.hasOwnProperty.call(seen, sig)) {
      const existing = out[seen[sig]];
      if (existing && contentId && !existing.contentId) existing.contentId = String(contentId);
      return;
    }
    seen[sig] = out.length;
    totalBytes += size;
    out.push({
      name: String(name || 'Email image'),
      source: String(source || 'INLINE'),
      contentId: String(contentId || ''),
      contentType: type,
      size: size,
      bytes: bytes
    });
  }

  function addAttachment_(att, source) {
    if (!att) return;
    try {
      const name = String(att.getName ? att.getName() : 'Email image');
      addBytes_(att.getBytes ? att.getBytes() : [], att.getContentType ? att.getContentType() : '', name, source);
    } catch (e) {}
  }

  // Collect real attachments first: the BR screenshot must not lose a slot to
  // the many small signature images often carried by forwarded mail.
  try {
    (msg.getAttachments({includeInlineImages:false, includeAttachments:true}) || []).forEach(function(att) {
      addAttachment_(att, 'ATTACHMENT');
    });
  } catch (e) {}

  try {
    (msg.getAttachments({includeInlineImages:true, includeAttachments:false}) || []).forEach(function(att) {
      addAttachment_(att, 'INLINE');
    });
  } catch (e) {}

  // Some Outlook-originated inline images only remain visible in Gmail raw MIME.
  try {
    const lines = String(msg.getRawContent() || '').replace(/\r/g, '').split('\n');
    for (let i = 0; i < lines.length; i++) {
      const typeMatch = String(lines[i] || '').match(/^Content-Type:\s*(image\/[a-z0-9.+-]+)/i);
      if (!typeMatch) continue;
      const headers = [lines[i]];
      let j = i + 1;
      while (j < lines.length && lines[j] !== '') { headers.push(lines[j]); j++; }
      const headerText = headers.join('\n');
      if (!/Content-Transfer-Encoding:\s*base64/i.test(headerText) || j >= lines.length) continue;
      const nameMatch = headerText.match(/(?:name|filename)\*?=\s*(?:"([^"]+)"|([^;\n]+))/i);
      const contentIdMatch = headerText.match(/^Content-ID:\s*<?([^>\s]+)>?/im);
      const name = nameMatch ? String(nameMatch[1] || nameMatch[2] || 'Pasted image').trim() : 'Pasted image';
      const encoded = [];
      j++;
      while (j < lines.length && !/^--/.test(lines[j])) {
        const line = String(lines[j] || '').trim();
        if (line && /^[A-Za-z0-9+\/=]+$/.test(line)) encoded.push(line);
        j++;
      }
      if (encoded.length) {
        const source = /Content-Disposition:\s*attachment/i.test(headerText) ? 'ATTACHMENT' : 'INLINE';
        try { addBytes_(Utilities.base64Decode(encoded.join('')), typeMatch[1], name, source, contentIdMatch ? contentIdMatch[1] : ''); } catch (e) {}
      }
      i = Math.max(i, j - 1);
    }
  } catch (e) {}

  return out;
}

function routingImagePayload_(parts) {
  return (parts || []).map(function(part) {
    return {
      name: part.name,
      source: part.source,
      contentId: part.contentId || '',
      contentType: part.contentType,
      size: part.size,
      dataUrl: 'data:' + part.contentType + ';base64,' + Utilities.base64Encode(part.bytes)
    };
  });
}

// V8.1.10: preserve Gmail's original message order. Inline images are written back
// into their original <img src="cid:..."> positions instead of being rendered
// as a separate image gallery above the message.
function routingSafeMessageHtml_(msg, imageParts, imagePayload) {
  let html = '';
  try { html = String(msg.getBody() || ''); } catch (e) { html = ''; }
  if (!html) return '';

  const images = imagePayload || routingImagePayload_(imageParts || []);
  const byContentId = {};
  images.forEach(function(image) {
    const key = String(image.contentId || '').replace(/^cid:/i, '').replace(/[<>]/g, '').trim().toLowerCase();
    if (key) byContentId[key] = image;
  });

  html = html.replace(/(<img\b[^>]*\bsrc\s*=\s*)(["'])cid:([^"']+)\2/gi, function(match, prefix, quote, rawContentId) {
    let contentId = String(rawContentId || '').replace(/[<>]/g, '').trim().toLowerCase();
    try { contentId = decodeURIComponent(contentId); } catch (e) {}
    const image = byContentId[contentId] || null;
    if (image) image._usedInBody = true;
    return image && image.dataUrl ? prefix + quote + image.dataUrl + quote : match;
  });

  // Remove active/embedded content while keeping Gmail's normal tables,
  // spacing, typography and image placement intact.
  html = html
    .replace(/<(?:script|iframe|object|embed|form|input|button|meta|link|base)\b[^>]*>[\s\S]*?<\/(?:script|iframe|object|embed|form|button)>/gi, '')
    .replace(/<(?:script|iframe|object|embed|form|input|button|meta|link|base)\b[^>]*\/?\s*>/gi, '')
    .replace(/<style\b[^>]*>[\s\S]*?<\/style>/gi, '')
    .replace(/\s+on[a-z]+\s*=\s*(?:"[^"]*"|'[^']*'|[^\s>]+)/gi, '')
    .replace(/\s+srcdoc\s*=\s*(?:"[^"]*"|'[^']*'|[^\s>]+)/gi, '')
    .replace(/\b(href|src)\s*=\s*(["'])\s*javascript:[\s\S]*?\2/gi, '$1="#"')
    .replace(/<img\b[^>]*\bsrc\s*=\s*(["'])cid:[^"']+\1[^>]*>/gi, '');

  html = html.replace(/<a\b([^>]*)>/gi, function(match, attrs) {
    attrs = String(attrs || '').replace(/\s+(?:target|rel)\s*=\s*(?:"[^"]*"|'[^']*'|[^\s>]+)/gi, '');
    return '<a' + attrs + ' target="_blank" rel="noopener noreferrer">';
  });

  // Gmail also exposes pasted screenshots as ordinary attachments. Show every
  // collected image that was not already embedded, including those attachments.
  // Never truncate: slicing a base64 image produces a blank image.
  const unmatched = images.filter(function(image) { return !image._usedInBody && image.dataUrl; });
  if (unmatched.length) {
    html += '<div class="mail-unmatched-inline-images">' + unmatched.map(function(image) {
      const safeName = String(image.name || 'Email image')
        .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
      return '<div class="mail-unmatched-inline-image"><img src="' + image.dataUrl + '" alt="' + safeName + '"></div>';
    }).join('') + '</div>';
  }
  return html;
}

function routingJoinBytes_(chunks) {
  const out = [];
  (chunks || []).forEach(function(chunk) {
    const bytes = chunk || [];
    for (let i = 0; i < bytes.length; i++) out.push(bytes[i]);
  });
  return out;
}

// V7.28.2: Drive advanced service remains the fast OCR path, but the Workbench
// now falls back to the Drive REST API with the Apps Script OAuth token. This
// prevents a missing Drive service toggle from disabling screenshot/PDF OCR.
function routingCreateConvertedFileRest_(part, targetMimeType, useOcr) {
  const boundary = '-------routing_' + new Date().getTime() + '_' + Math.floor(Math.random() * 1000000);
  const metadata = JSON.stringify({
    name: 'Routing OCR ' + new Date().getTime(),
    mimeType: targetMimeType
  });
  const before = Utilities.newBlob(
    '--' + boundary + '\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n' +
    metadata + '\r\n--' + boundary + '\r\nContent-Type: ' + part.contentType + '\r\n\r\n'
  ).getBytes();
  const after = Utilities.newBlob('\r\n--' + boundary + '--').getBytes();
  const url = 'https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&fields=id' +
    (useOcr ? '&ocrLanguage=en' : '');
  const response = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'multipart/related; boundary=' + boundary,
    headers: {Authorization: 'Bearer ' + ScriptApp.getOAuthToken()},
    payload: routingJoinBytes_([before, part.bytes || [], after]),
    muteHttpExceptions: true
  });
  const code = Number(response.getResponseCode() || 0);
  const body = String(response.getContentText() || '');
  if (code < 200 || code >= 300) {
    let message = body;
    try { message = JSON.parse(body).error.message || body; } catch (e) {}
    throw new Error('Drive OCR ' + code + ': ' + String(message || 'request failed'));
  }
  let result = {};
  try { result = JSON.parse(body); } catch (e) {}
  return String(result.id || '');
}

function routingDeleteTemporaryDriveFile_(fileId) {
  if (!fileId) return;
  try {
    if (typeof Drive !== 'undefined' && Drive.Files && Drive.Files.remove) {
      Drive.Files.remove(fileId);
      return;
    }
  } catch (e) {}
  try { DriveApp.getFileById(fileId).setTrashed(true); return; } catch (e) {}
  try {
    UrlFetchApp.fetch('https://www.googleapis.com/drive/v3/files/' + encodeURIComponent(fileId), {
      method: 'delete',
      headers: {Authorization: 'Bearer ' + ScriptApp.getOAuthToken()},
      muteHttpExceptions: true
    });
  } catch (e) {}
}

function routingOcrOneImage_(part) {
  let fileId = '';
  let advancedError = '';
  const blob = Utilities.newBlob(part.bytes, part.contentType, part.name || 'routing-evidence');

  if (typeof Drive !== 'undefined' && Drive.Files && Drive.Files.create) {
    try {
      const file = Drive.Files.create({
        name: 'Routing OCR ' + new Date().getTime(),
        mimeType: 'application/vnd.google-apps.document'
      }, blob, {ocrLanguage:'en', fields:'id'});
      fileId = String(file && file.id || '');
      if (fileId) return String(DocumentApp.openById(fileId).getBody().getText() || '');
    } catch (e) {
      advancedError = String(e && e.message || e || '');
    } finally {
      routingDeleteTemporaryDriveFile_(fileId);
      fileId = '';
    }
  }

  try {
    fileId = routingCreateConvertedFileRest_(part, 'application/vnd.google-apps.document', true);
    if (!fileId) throw new Error('Drive OCR did not return a document ID.');
    return String(DocumentApp.openById(fileId).getBody().getText() || '');
  } catch (e) {
    const restError = String(e && e.message || e || 'OCR unavailable');
    throw new Error(advancedError ? (restError + ' (advanced service also failed: ' + advancedError + ')') : restError);
  } finally {
    routingDeleteTemporaryDriveFile_(fileId);
  }
}

      const name = String(att.getName ? att.getName() : 'Attachment.pdf');
      let contentType = String(att.getContentType ? att.getContentType() : '').toLowerCase().split(';')[0];
      if (!contentType && /\.pdf$/i.test(name)) contentType = 'application/pdf';
      if (contentType !== 'application/pdf') return;

      const size = Number(bytes.length || 0);
      const key = name + '|' + size;
      if (!size || size > 8000000 || seen[key] || out.length >= 8) return;
      seen[key] = true;
      out.push({name:name, source:'ATTACHMENT', contentType:contentType, size:size, bytes:bytes});
    });
  } catch (e) {}
  return out;
}

function routingOcrMessageImages_(msg, suppliedParts, imagesOnly) {
  const messageId = String(msg && msg.getId ? msg.getId() : '');
  const key = (imagesOnly ? 'routing_ocr_v815_images_' : 'routing_ocr_v814_') + messageId;
  const cache = CacheService.getUserCache();
  const cached = messageId ? cache.get(key) : '';
  if (cached) { try { return JSON.parse(cached); } catch (e) {} }

  const parts = imagesOnly ? (suppliedParts || routingCollectMessageImages_(msg)) : routingCollectMessageOcrParts_(msg, suppliedParts);
  const reviewParts = parts.filter(function(part) {
    return part.size >= 12000 && (imagesOnly
      ? /^image\/(?:png|jpeg|gif|bmp|webp)$/i.test(part.contentType)
      : /^(?:image\/(?:png|jpeg|gif|bmp|webp)|application\/pdf)$/i.test(part.contentType));
  }).slice(0, 4);
  const texts = [];
  const evidence = [];
  const prelistNoRoutingIds = [];
  let error = '';

  reviewParts.forEach(function(part) {
    try {
      const text = routingOcrOneImage_(part);
      if (text) {
        texts.push(text);
        extractPrelistNoRoutingPartsFromOcr_(text).forEach(function(id){
          if (prelistNoRoutingIds.indexOf(id) < 0) prelistNoRoutingIds.push(id);
        });
        const ids = extractRoutingBrIdsLoose_(text);
        if (ids.length) evidence.push({
          name: String(part.name || 'Email file'),
          source: String(part.source || ''),
          contentType: String(part.contentType || ''),
          size: Number(part.size || 0),
          imageIndex: /^image\//i.test(part.contentType) ? (suppliedParts || parts).indexOf(part) : -1,
          ids: ids
        });
      }
    } catch (e) {
      if (!error) error = String(e && e.message || e || 'OCR unavailable');
    }
  });

  const result = {
    text: texts.join('\n'),
    imageCount: reviewParts.length,
    ocrImageCount: texts.length,
    evidence: evidence,
    prelistNoRoutingIds: prelistNoRoutingIds,
    error: error
  };
  if (messageId && !error) { try { cache.put(key, JSON.stringify(result), 21600); } catch (e) {} }
  return result;
}

function extractRoutingBrIdsLoose_(text) {
  const out = [], seen = {};
  const normalized = String(text || '').toUpperCase()
    .replace(/\bB\s+R\s*[-:#]?\s*(?=[0-9OIL])/g, 'BR')
    .replace(/\bBR\s*[-:#]?\s*(?=[0-9OIL])/g, 'BR');
  const parts = normalized.match(/\bBR[0-9OIL]{6,}\b/g) || [];
  parts.forEach(function(part) {
    const fixed = 'BR' + part.substring(2).replace(/O/g, '0').replace(/[IL]/g, '1');
    if (!seen[fixed]) { seen[fixed] = true; out.push(fixed); }
  });
  return out;
}

// Read a BR description from the same OCR row as the object number. FINAL
// screenshots usually contain rows such as:
//   0012  BR1379065  FLIPPER ARM ASSY.  1 PCS ...
// Keep the text column after the BR number and stop before quantity/unit data.
function extractRoutingDescriptions_(text) {
  const out = {};
  const source = String(text || '').replace(/\r/g, '');
  const matches = [];
  const re = /\bB\s*R\s*[-:#]?\s*([0-9OIL]{6,})\b/ig;
  let match;
  while ((match = re.exec(source)) !== null) matches.push({match:match, index:match.index, end:re.lastIndex});
  matches.forEach(function(item, index) {
    const idMatch = item.match;
    const part = 'BR' + String(idMatch[1] || '').toUpperCase().replace(/O/g, '0').replace(/[IL]/g, '1');
    if (!/^BR\d{6,}$/.test(part)) return;
    const nextIndex = index + 1 < matches.length ? matches[index + 1].index : source.length;
    let tail = source.substring(item.end, nextIndex)
      .replace(/\n+/g, ' ')
      .replace(/^[\s|:;,.\-–—]+/, '')
      .trim();
    if (!tail) return;
    let description = tail;
    description = description
      .replace(/\s+\d+(?:[.,]\d+)?\s*(?:PCS?|EA|SET|UNIT|UNITS)\b[\s\S]*$/i, '')
      .replace(/\s+\.{2,}\d+\s+\d{3,4}\b[\s\S]*$/i, '')
      .replace(/\s+\d+\s+\d{3,4}\s+(?=[A-Z])/i, '')
      .replace(/\s+(?:MY\d{1,3}|F\d{2,4}|Z\d{1,3})\b[\s\S]*$/i, '')
      .replace(/^[\s|:;,.\-–—]+|[\s|:;,.\-–—]+$/g, '')
      .replace(/\s+/g, ' ')
      .trim();
    if (!description || description.length < 3 || !/[A-Z]/i.test(description)) return;
    if (/^(?:HAS BEEN|RELEASED|ASSIGNED|NO ROUTING|ROUTING|FINAL|PRELIST)\b/i.test(description)) return;
    if (!out[part] || description.length > out[part].length) out[part] = description.substring(0, 160);
  });
  return out;
}

function extractPrelistNoRoutingPartsFromEvidence_(text) {
  const lines = String(text || '').replace(/\r/g, '').split('\n');
  const out = [], seen = {};
  const noRoutingLabel = /\b(?:no\s*(?:routing|route)|not\s+routed|routing\s+(?:missing|blank|not\s+set))\b/i;
  function add_(value) {
    const part = String(value || '').toUpperCase();
    if (part && !seen[part]) { seen[part] = true; out.push(part); }
  }

  let underHeading = false;
  lines.forEach(function(line, index) {
    const clean = String(line || '').replace(/^\s*>+\s*/, '').trim();
    const ids = extractRoutingBrIdsLoose_(clean);
    const explicit = noRoutingLabel.test(clean);
    if (explicit && !ids.length) { underHeading = true; return; }
    if (underHeading && !clean) { underHeading = false; return; }
    if (underHeading && /^[A-Za-z][A-Za-z\s]{3,}:?\s*$/.test(clean) && !ids.length) underHeading = false;
    const blankRouting = ids.length && /(?:\||\t|,|;|\s)(?:-|–|—|n\/?a|none|blank|not\s+set)\s*$/i.test(clean);
    if (explicit) {
      if (ids.length) ids.forEach(add_);
      else if (index + 1 < lines.length) extractRoutingBrIdsLoose_(lines[index + 1]).forEach(add_);
    } else if (underHeading || blankRouting) {
      ids.forEach(add_);
    }
  });

  const all = extractRoutingBrIdsLoose_(text);
  // A pasted PRELIST screenshot is often only a compact Object-ID list without
  // a repeated NO ROUTING label. Treat it as a list only when most useful lines
  // are BR rows, which avoids harvesting unrelated numbers from normal prose.
  if (!out.length && all.length && !noRoutingLabel.test(text)) {
    const meaningful = lines.map(function(line) {
      return String(line || '').replace(/^\s*>+\s*/, '').trim();
    }).filter(function(line) { return line && !/^[-=*_.\s]+$/.test(line); });
    const idLineCount = meaningful.filter(function(line) { return extractRoutingBrIdsLoose_(line).length > 0; }).length;
    if (idLineCount >= 2 && idLineCount / Math.max(1, meaningful.length) >= 0.55) all.forEach(add_);
  }
  return out;
}

function extractPrelistNoRoutingPartsFromOcr_(text) {
  const lines = String(text || '').replace(/\r/g, '').split('\n');
  for (let i = 0; i < lines.length; i++) {
    const columns = lines[i].trim().split(/\t|\s{2,}|\s*\|\s*/);
    const column = columns.findIndex(function(cell){return /^no\s*(?:routing|route)$/i.test(cell.trim());});
    if (column < 0 || columns.length < 2) continue;
    const found = [], seen = {};
    for (let j = i + 1; j < lines.length; j++) {
      if (!lines[j].trim()) break;
      const cells = lines[j].trim().split(/\t|\s{2,}|\s*\|\s*/);
      extractRoutingBrIdsLoose_(cells[column] || '').forEach(function(id){if(!seen[id]){seen[id]=true;found.push(id);}});
    }
    if (found.length) return found;
  }
  return extractPrelistNoRoutingPartsFromEvidence_(text);
}

function extractFinalHasmanizamObjectIds_(text) {
  const out = [], seen = {};
  const lines = routingNewMessageText_(text).split(/\r?\n/);
  lines.forEach(function(line) {
    const t = String(line || '');
    if (/\bGA\s*#/i.test(t) || /\bSO\s*#/i.test(t) || /\bECO\b/i.test(t)) return;
    const parts = t.toUpperCase().match(/\bBR\d{6,}\b/g) || [];
    parts.forEach(function(p){ if (!seen[p]) { seen[p] = true; out.push(p); } });
  });
  return out;
}

function routingPhaseState_(receivedCount, doneCount, total, inProgressBy) {
  total = Math.max(1, Number(total || 0));
  if (doneCount >= total) return 'DONE';
  if (doneCount > 0) return 'PARTIAL';
  if (inProgressBy) return 'IN PROGRESS';
  if (receivedCount > 0) return receivedCount >= total ? 'RECEIVED' : 'PARTIAL';
  return 'NOT RECEIVED';
}

function buildRoutingPhaseSummary_(requests, targetSns) {
  const targetMap = {};
  (targetSns || []).forEach(function(sn){ const k=normalizeSn_(sn); if(k) targetMap[k]=true; });
  const total = Math.max(1, Object.keys(targetMap).length);
  function one(type) {
    const received = {}, done = {}, doneBy = {}, progressBy = {};
    let noRoutingCount = 0, listCount = 0, evidenceCount = 0, imageReview = 0, hasHasmanizam = false, ocrError = '';
    (requests || []).filter(function(r){return String(r.type||'').toUpperCase()===type;}).forEach(function(r){
      const matched = (r.sns || []).map(normalizeSn_).filter(function(k){return targetMap[k];});
      matched.forEach(function(k){ received[k]=true; });
      if (String(r.completionState||'') === 'DONE') matched.forEach(function(k){ done[k]=true; });
      if (r.completionBy && String(r.completionState||'') === 'DONE') doneBy[r.completionBy]=true;
      if (r.completionBy && String(r.completionState||'') === 'IN PROGRESS') progressBy[r.completionBy]=true;
      if (type === 'PRELIST') {
        noRoutingCount += (r.prelistNoRoutingParts || []).length;
        imageReview += Number(r.prelistImageReviewCount || 0);
        evidenceCount += Number(r.evidenceAttachmentCount || r.attachmentCount || 0) + Number(r.prelistImageReviewCount || 0);
        if (!ocrError && r.prelistOcrError) ocrError = r.prelistOcrError;
      } else {
        listCount += (r.finalHasmanizamParts || []).length;
        imageReview += Number(r.finalHasmanizamImageReviewCount || 0);
        if (r.finalHasmanizamSeen) hasHasmanizam = true;
        if (!ocrError && r.finalHasmanizamOcrError) ocrError = r.finalHasmanizamOcrError;
      }
    });
    const rCount=Object.keys(received).length, dCount=Object.keys(done).length;
    const progressNames=Object.keys(progressBy), doneNames=Object.keys(doneBy);
    let state=routingPhaseState_(rCount,dCount,total,progressNames.join(' / '));
    const replyDone=dCount>=total;
    let autoClear=false, actionState='WAITING', listState='';
    if(type==='PRELIST'){
      autoClear=rCount>=total && noRoutingCount===0 && evidenceCount===0 && !ocrError;
      if(replyDone) actionState='DONE';
      else if(autoClear) actionState='CLEAR';
      else if(rCount===0) actionState='WAITING';
      else if(noRoutingCount>0) actionState='TO DO';
      else if(ocrError || evidenceCount>0) actionState='REVIEW';
      else actionState='WAITING';
    }else{
      autoClear=rCount>=total && hasHasmanizam && listCount===0 && imageReview===0 && !ocrError;
      if(replyDone) actionState='DONE';
      else if(autoClear) actionState='CLEAR';
      else if(rCount===0 || !hasHasmanizam) actionState='WAITING';
      else if(listCount>0) actionState='TO DO';
      else if(ocrError || imageReview>0) actionState='REVIEW';
      else actionState='WAITING';
      if(autoClear) listState='CLEAR';
      else if(listCount>0) listState='LIST READY';
      else if(ocrError || imageReview>0) listState='IMAGE REVIEW';
      else if(rCount>0 && !hasHasmanizam) listState='WAITING HASMANIZAM';
    }
    if(autoClear && !replyDone) state='DONE';
    return {
      state:state,total:total,receivedCount:rCount,doneCount:dCount,
      doneBy:doneNames.join(' / '),inProgressBy:progressNames.join(' / '),
      noRoutingCount:noRoutingCount,listCount:listCount,evidenceCount:evidenceCount,imageReviewCount:imageReview,
      hasHasmanizam:hasHasmanizam,autoClear:autoClear,actionState:actionState,listState:listState,ocrError:ocrError
    };
  }
  return {prelist:one('PRELIST'), final:one('FINAL')};
}

function routingPhaseCacheKey_(groups) {
  const raw = (groups || []).map(function(g){
    return String(g.key||'')+'|'+(g.sns||[]).map(normalizeSn_).filter(Boolean).sort().join(',');
  }).sort().join('||');
  const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, raw, Utilities.Charset.UTF_8);
  return 'routing_phase_v8114_' + Utilities.base64EncodeWebSafe(digest).replace(/=+$/,'').substring(0,80);
}

function getRoutingEmailRequestsForGroup(sns, forceRefresh) {
  const rawList = Array.isArray(sns) ? sns : [sns];
  const cleanList = [];
  const targetMap = {};

  rawList.forEach(function(sn) {
    const display = String(sn || '').trim();
    const key = normalizeSn_(display);
    if (!key || targetMap[key]) return;
    targetMap[key] = true;
    cleanList.push(display);
  });

  if (!cleanList.length) {
    return {
      sn: '',
      sns: [],
      ok: true,
      requests: [],
      latest: null,
      prelist: null,
      final: null,
      requiredRoutingItems: [],
      searchedAt: formatDateTime_(new Date())
    };
  }

  const cache = CacheService.getScriptCache();
  const cacheKey = 'routing_mail_v819_' + cleanList
    .map(normalizeSn_)
    .filter(Boolean)
    .sort()
    .join('_')
    .substring(0, 180);

  if (!forceRefresh) {
    const cached = cache.get(cacheKey);
    if (cached) {
      try {
        return JSON.parse(cached);
      } catch (e) {}
    }
  }

  // V7.22 FAST PATH: search by the selected Engineer design-group S/N first.
  // Previous versions scanned hundreds of generic Routing threads for EVERY group click.
  // This narrows Gmail to only threads containing one of the selected S/N values, then
  // the existing classifier below still verifies PRELIST / FINAL and exact S/N matches.
  const threadMap = {};
  const searchSns = cleanList
    .map(normalizeSn_)
    .filter(Boolean)
    .filter(function(value, index, arr) { return arr.indexOf(value) === index; });

  function addRoutingSearchThreads_(query, maxResults) {
    let threads = [];
    try {
      threads = GmailApp.search(query, 0, maxResults || 50);
    } catch (e) {
      throw new Error(
        'Gmail access is not authorized yet. Run testGmailConnection() once in Apps Script and allow Gmail access.'
      );
    }
    threads.forEach(function(thread) {
      threadMap[thread.getId()] = thread;
    });
  }

  // Gmail supports braces as OR. Chunk the list so large Engineer groups do not
  // create an excessively long query. Most groups now need only ONE Gmail search.
  for (let i = 0; i < searchSns.length; i += 12) {
    const chunk = searchSns.slice(i, i + 12);
    const terms = chunk.map(function(sn) {
      return '"' + String(sn).replace(/"/g, '') + '"';
    }).join(' ');
    addRoutingSearchThreads_('newer_than:730d {' + terms + '}', 60);
  }

  // Rare fallback: if Gmail did not return anything for the S/N query (for example
  // an unusual subject/range format), do a much smaller generic Routing scan.
  // This is intentionally only used when the fast path returns zero candidate threads.
  if (!Object.keys(threadMap).length) {
    addRoutingSearchThreads_('newer_than:730d "Planning/Purchasing"', 50);
    addRoutingSearchThreads_('newer_than:730d "Final Drawing"', 50);
  }

  const found = [];

  // One Gmail THREAD = one Routing request card.
  // Match against any S/N in the selected Engineer design group.
  Object.keys(threadMap).forEach(function(id) {
    const thread = threadMap[id];
    const permalink = thread.getPermalink();
    const messages = thread.getMessages();

    const matchedMessages = [];

    messages.forEach(function(msg) {
      if (isRoutingHiddenSender_(String(msg.getFrom() || ''))) return;
      const subject = String(msg.getSubject() || '').trim();
      const plainBody = String(msg.getPlainBody() || '').trim();
      const combined = subject + '\n' + plainBody;

      const type = classifyRoutingRequestEmail_(subject, plainBody);
      if (!type) return;

      const extractedSns = extractSnListFromRoutingEmail_(combined);
      const match = extractedSns.some(function(x) {
        return !!targetMap[normalizeSn_(x)];
      });

      if (!match) return;

      matchedMessages.push({
        message: msg,
        type: type,
        subject: subject,
        plainBody: plainBody,
        sns: extractedSns,
        date: msg.getDate()
      });
    });

    if (!matchedMessages.length) return;

    // A Gmail thread may contain the PRELIST request and a later FINAL request.
    // Keep one source card per phase so the user can inspect both side by side.
    const requestTypes = ['PRELIST', 'FINAL'].filter(function(type) {
      return matchedMessages.some(function(x) { return x.type === type; });
    });

    requestTypes.forEach(function(requestType) {
      const sameType = matchedMessages.filter(function(x) {
        return x.type === requestType;
      });

    const latestMatched = sameType
      .slice()
      .sort(function(a, b) {
        return b.date.getTime() - a.date.getTime();
      })[0];

    const canonical = latestMatched;
    const attachmentMap = {};
    const snMap = {};
    const prelistNoRoutingMap = {};
    const finalHasmanizamMap = {};
    const finalHasmanizamDescriptionMap = {};
    let finalEvidenceParts = [];
    let prelistImageReviewCount = 0;
    let finalHasmanizamImageReviewCount = 0;
    let finalHasmanizamSeen = false;
    let finalEvidenceMessage = null;
    let finalEvidenceRank = -1;
    let prelistOcrError = '';
    let finalHasmanizamOcrError = '';

    sameType.forEach(function(item) {
      (item.sns || []).forEach(function(x) {
        const k = normalizeSn_(x);
        if (k) snMap[k] = String(x);
      });

      const atts = item.message.getAttachments({
        includeInlineImages: false,
        includeAttachments: true
      }) || [];

      atts.forEach(function(att) {
        const name = String(att.getName() || 'Attachment');
        const contentType = String(att.getContentType() || '');
        const key = name + '|' + contentType;

        if (!attachmentMap[key]) {
          attachmentMap[key] = {
            name: name,
            contentType: contentType,
            size: Number(att.getBytes().length || 0)
          };
        }
      });

      if (requestType === 'PRELIST') {
        extractPrelistNoRoutingPartsFromAttachments_(atts).forEach(function(part) {
          prelistNoRoutingMap[String(part).toUpperCase()] = true;
        });
        extractPrelistNoRoutingPartsFromText_(routingNewMessageText_(item.plainBody)).forEach(function(part) {
          prelistNoRoutingMap[String(part).toUpperCase()] = true;
        });
        const prelistOcr = routingOcrMessageImages_(item.message);
        prelistImageReviewCount += Number(prelistOcr.imageCount || 0);
        (prelistOcr.prelistNoRoutingIds || extractPrelistNoRoutingPartsFromOcr_(prelistOcr.text)).forEach(function(part) {
          prelistNoRoutingMap[String(part).toUpperCase()] = true;
        });
        if (!prelistOcrError && prelistOcr.error) prelistOcrError = prelistOcr.error;
      }

      // FINAL numbers are collected below, exclusively from Hasmanizam's own
      // message text and its image screenshots.
    });

    if (requestType === 'FINAL') {
      messages.forEach(function(msg) {
        const sender = String(msg.getFrom() || '');
        if (!/hasmanizam/i.test(sender)) return;
        finalHasmanizamSeen = true;
        let body = '';
        try { body = String(msg.getPlainBody() || ''); } catch (e) { body = ''; }
        const bodyParts = extractFinalHasmanizamObjectIds_(body);
        const finalOcr = routingOcrMessageImages_(msg, null, true);
        finalHasmanizamImageReviewCount += Number(finalOcr.imageCount || 0);
        const ocrParts = extractRoutingBrIdsLoose_(finalOcr.text);
        const descriptions = extractRoutingDescriptions_(finalOcr.text);
        // Open the actual Hasmanizam message carrying the BR screenshot,
        // instead of combining every Hasmanizam screenshot in the thread.
        const rank = (finalOcr.evidence || []).some(function(e){return /^image\//i.test(e.contentType);}) ? 3 :
          (finalOcr.evidence || []).length ? 2 : (finalOcr.imageCount ? 1 : 0);
        if (rank > finalEvidenceRank || (rank === finalEvidenceRank && finalEvidenceMessage && msg.getDate() > finalEvidenceMessage.getDate())) {
          finalEvidenceMessage = msg;
          finalEvidenceRank = rank;
          finalEvidenceParts = (ocrParts.length ? ocrParts : bodyParts).slice();
          Object.keys(finalHasmanizamDescriptionMap).forEach(function(part) { delete finalHasmanizamDescriptionMap[part]; });
          Object.keys(descriptions).forEach(function(part) {
            const value = String(descriptions[part] || '').trim();
            if (value) finalHasmanizamDescriptionMap[String(part).toUpperCase()] = value;
          });
        }
        if (!finalHasmanizamOcrError && finalOcr.error) finalHasmanizamOcrError = finalOcr.error;
      });
      finalEvidenceParts.forEach(function(part) {
        finalHasmanizamMap[String(part).toUpperCase()] = true;
      });
    }

    const firstRequestDate = sameType.reduce(function(minDate, item) {
      if (!minDate || item.date.getTime() < minDate.getTime()) return item.date;
      return minDate;
    }, null);
    const completion = routingReplyStateForThread_(messages, firstRequestDate);

    const attachments = Object.keys(attachmentMap).map(function(k) {
      return attachmentMap[k];
    });

    const partNumbers = requestType === 'FINAL' ? Object.keys(finalHasmanizamMap).sort() : [];
    const prelistNoRoutingParts = Object.keys(prelistNoRoutingMap).sort();
    const finalHasmanizamParts = Object.keys(finalHasmanizamMap).sort();
    const requestRequiredItems = [];

    if (String(completion.state || '') !== 'DONE') {
      prelistNoRoutingParts.forEach(function(part) {
        requestRequiredItems.push({
          part: part,
          source: 'PRELIST',
          reason: 'NO ROUTING',
          status: 'NEED ROUTING'
        });
      });

      finalHasmanizamParts.forEach(function(part) {
        requestRequiredItems.push({
          part: part,
          source: 'FINAL',
          reason: 'HASMANIZAM LIST',
          status: 'NEED ROUTING'
        });
      });
    }

    const snsFound = Object.keys(snMap).map(function(k) {
      return snMap[k];
    });

    const activityDate = latestMatched.date;

      found.push({
      type: requestType,
      date: formatDateTime_(activityDate),
      dateMs: activityDate.getTime(),
      subject: canonical.subject,
      from: String(canonical.message.getFrom() || ''),
      permalink: permalink,
      threadId: thread.getId(),
      messageId: canonical.message.getId(),
      evidenceMessageId: requestType === 'FINAL' && finalEvidenceMessage ? finalEvidenceMessage.getId() : '',
      evidenceRank: requestType === 'FINAL' ? finalEvidenceRank : -1,
      messageCount: matchedMessages.length,
      matchedMessageCount: sameType.length,
      sns: snsFound,
      attachments: attachments,
      attachmentCount: attachments.length,
      partNumbers: partNumbers,
      partCount: partNumbers.length,
      prelistNoRoutingParts: prelistNoRoutingParts,
      finalHasmanizamParts: finalHasmanizamParts,
      finalHasmanizamDescriptions: finalHasmanizamDescriptionMap,
      prelistImageReviewCount: prelistImageReviewCount,
      finalHasmanizamImageReviewCount: finalHasmanizamImageReviewCount,
      finalHasmanizamSeen: finalHasmanizamSeen,
      prelistOcrError: prelistOcrError,
      finalHasmanizamOcrError: finalHasmanizamOcrError,
      completionState: completion.state || '',
      completionBy: completion.by || '',
      completionAt: completion.atText || '',
      routingRequiredItems: requestRequiredItems,
      preview: compactEmailPreview_(canonical.plainBody)
      });
    });
  });

  found.sort(function(a, b) {
    return Number(b.dateMs || 0) - Number(a.dateMs || 0);
  });

  const mergedRequests = mergeRoutingRequestCards_(found);
  const latestPrelist = mergedRequests.find(function(x) { return x.type === 'PRELIST'; }) || null;
  const latestFinal = mergedRequests.find(function(x) { return x.type === 'FINAL'; }) || null;

  const requiredMap = {};
  const detectedMap = {};
  const descriptionMap = {};

  mergedRequests.forEach(function(req) {
    Object.keys(req.finalHasmanizamDescriptions || {}).forEach(function(part) {
      const key = String(part || '').toUpperCase().trim();
      const value = String(req.finalHasmanizamDescriptions[part] || '').trim();
      if (key && value && (!descriptionMap[key] || value.length > descriptionMap[key].length)) descriptionMap[key] = value;
    });
    (req.routingRequiredItems || []).forEach(function(item) {
      const part = String(item.part || '').toUpperCase().trim();
      if (!part) return;
      if (!requiredMap[part]) requiredMap[part] = { part:part, sources:[], reasons:[] };
      const source = String(item.source || '').trim();
      const reason = String(item.reason || '').trim();
      if (source && requiredMap[part].sources.indexOf(source) < 0) requiredMap[part].sources.push(source);
      if (reason && requiredMap[part].reasons.indexOf(reason) < 0) requiredMap[part].reasons.push(reason);
    });

    // Keep OCR/text detections available even when the request thread already
    // contains a colleague's Done reply. The phase may be DONE while the user
    // still needs to inspect or hand off the detected IDs.
    const detected = String(req.type || '').toUpperCase() === 'PRELIST'
      ? (req.prelistNoRoutingParts || [])
      : (req.finalHasmanizamParts || []);
    detected.forEach(function(part) {
      const p = String(part || '').toUpperCase().trim();
      if (!p) return;
      if (!detectedMap[p]) detectedMap[p] = {part:p, source:String(req.type || '').toUpperCase(), reason:String(req.type || '').toUpperCase() === 'FINAL' ? 'HASMANIZAM LIST' : 'NO ROUTING'};
    });
  });

  const requiredRoutingItems = Object.keys(requiredMap).map(function(part) {
    const x = requiredMap[part];
    x.sources.sort(function(a,b){ if(a === b) return 0; if(a === 'PRELIST') return -1; if(b === 'PRELIST') return 1; return a.localeCompare(b); });
    return {
      part: x.part,
      source: x.sources.join(' + '),
      reason: x.reasons.join(' / '),
      status: 'NEED ROUTING'
    };
  }).sort(function(a,b){ return a.part.localeCompare(b.part); });

  const detectedRoutingItems = Object.keys(detectedMap).map(function(part) {
    return {
      part: detectedMap[part].part,
      source: detectedMap[part].source,
      reason: detectedMap[part].reason,
      status: 'DETECTED'
    };
  }).sort(function(a,b){ return a.part.localeCompare(b.part); });

  const result = {
    sn: cleanList[0] || '',
    sns: cleanList,
    ok: true,
    requests: mergedRequests.slice(0, 10),
    latest: mergedRequests.length ? mergedRequests[0] : null,
    prelist: latestPrelist,
    final: latestFinal,
    phaseSummary: buildRoutingPhaseSummary_(mergedRequests, cleanList),
    requiredRoutingItems: requiredRoutingItems,
    detectedRoutingItems: detectedRoutingItems,
    routingDescriptions: descriptionMap,
    searchedAt: formatDateTime_(new Date())
  };

  try {
    cache.put(cacheKey, JSON.stringify(result), 600);
  } catch (e) {}

  return result;
}

function routingRequestSemanticKey_(req) {
  req = req || {};

  const type = String(req.type || '').toUpperCase();
  const subject = String(req.subject || '');

  // V7.8: ECO is the strongest business-request identity.
  // If two cards have the same Routing phase + ECO, treat them as the same
  // request even when forwarded/quoted history causes a slightly different
  // extracted S/N set. This is what removes duplicate Supplementary/Fwd cards.
  const ecoMatch = subject.match(/\bECO\d*[-_ ]?\d+\b/i);
  const eco = ecoMatch
    ? ecoMatch[0].toUpperCase().replace(/\s+/g, '')
    : '';

  if (eco) return [type, eco].join('|');

  const sns = (req.sns || [])
    .map(normalizeSn_)
    .filter(Boolean)
    .sort()
    .join(',');

  // No ECO: fall back to a cleaned subject + S/N set.
  const cleanedSubject = subject
    .replace(/^\s*(?:fwd?|re)\s*:\s*/ig, '')
    .replace(/supplementary\s+summary\s*:\s*/ig, '')
    .replace(/\s+/g, ' ')
    .trim()
    .toUpperCase();

  return [type, cleanedSubject, sns].join('|');
}

function mergeRoutingRequestCards_(requests) {
  const groups = {};

  (requests || []).forEach(function(req) {
    const key = routingRequestSemanticKey_(req);
    if (!groups[key]) groups[key] = [];
    groups[key].push(req);
  });

  const out = [];

  Object.keys(groups).forEach(function(key) {
    const items = groups[key]
      .slice()
      .sort(function(a, b) {
        return Number(b.dateMs || 0) - Number(a.dateMs || 0);
      });

    // Automated sgase notifications can establish the workflow/thread, but
    // must never become the email shown as the user's PRELIST/FINAL reference.
    const visibleReference = items.find(function(item) {
      return !isRoutingReferenceExcludedSender_(item.from || '');
    });
    const latest = Object.assign({}, visibleReference || items[0]);
    latest.dedupedCardCount = items.length;

    // Preserve Routing setup findings from every duplicated/forwarded copy,
    // while still showing only the latest email card.
    const reqSeen = {};
    const reqItems = [];
    const preSeen = {};
    const finalSeen = {};
    const finalDescriptions = {};
    items.forEach(function(item) {
      (item.routingRequiredItems || []).forEach(function(r) {
        const part = String(r.part || '').toUpperCase().trim();
        const k = part + '|' + String(r.source || '') + '|' + String(r.reason || '');
        if (!part || reqSeen[k]) return;
        reqSeen[k] = true;
        reqItems.push({part:part, source:r.source || '', reason:r.reason || '', status:'NEED ROUTING'});
      });
      (item.prelistNoRoutingParts || []).forEach(function(part) { preSeen[String(part).toUpperCase()] = true; });
      (item.finalHasmanizamParts || []).forEach(function(part) { finalSeen[String(part).toUpperCase()] = true; });
      Object.keys(item.finalHasmanizamDescriptions || {}).forEach(function(part) {
        const key = String(part || '').toUpperCase();
        const value = String(item.finalHasmanizamDescriptions[part] || '').trim();
        if (value && (!finalDescriptions[key] || value.length > finalDescriptions[key].length)) finalDescriptions[key] = value;
      });
    });
    latest.routingRequiredItems = reqItems;
    latest.prelistNoRoutingParts = Object.keys(preSeen).sort();
    latest.finalHasmanizamParts = Object.keys(finalSeen).sort();
    latest.finalHasmanizamDescriptions = finalDescriptions;
    latest.prelistImageReviewCount = items.reduce(function(n,x){return n+Number(x.prelistImageReviewCount||0);},0);
    latest.finalHasmanizamImageReviewCount = items.reduce(function(n,x){return n+Number(x.finalHasmanizamImageReviewCount||0);},0);
    latest.evidenceAttachmentCount = items.reduce(function(n,x){return n+Number(x.attachmentCount||0);},0);
    latest.finalHasmanizamSeen = items.some(function(x){return !!x.finalHasmanizamSeen;});
    const sourceItem = items.filter(function(x){return x.evidenceMessageId;}).sort(function(a,b){return Number(b.evidenceRank||0)-Number(a.evidenceRank||0)||Number(b.dateMs||0)-Number(a.dateMs||0);})[0];
    if(sourceItem){latest.evidenceMessageId=sourceItem.evidenceMessageId;latest.threadId=sourceItem.threadId;}
    const preOcrFailed = items.find(function(x){return !!x.prelistOcrError;});
    const finalOcrFailed = items.find(function(x){return !!x.finalHasmanizamOcrError;});
    latest.prelistOcrError = preOcrFailed ? preOcrFailed.prelistOcrError : '';
    latest.finalHasmanizamOcrError = finalOcrFailed ? finalOcrFailed.finalHasmanizamOcrError : '';
    const completionItems = items.filter(function(x){return x.completionState;}).sort(function(a,b){return Number(b.dateMs||0)-Number(a.dateMs||0);});
    const doneItem = completionItems.find(function(x){return x.completionState==='DONE';});
    const comp = doneItem || completionItems[0];
    if(comp){latest.completionState=comp.completionState||'';latest.completionBy=comp.completionBy||'';latest.completionAt=comp.completionAt||'';}
    if(latest.completionState==='DONE') latest.routingRequiredItems=[];

    out.push(latest);
  });

  out.sort(function(a, b) {
    return Number(b.dateMs || 0) - Number(a.dateMs || 0);
  });

  return out;
}

function classifyRoutingRequestEmail_(subject, body) {
  const s = String(subject || '');
  const b = String(body || '');
  const text = (s + '\n' + b).replace(/\s+/g, ' ');

  const prelist =
    /release\s+of\s+bom(?:\(s\))?\s+for\s+planning\s*\/\s*purchasing/i.test(text) ||
    (/bom(?:\(s\))?/i.test(text) &&
     /planning\s*\/\s*purchasing/i.test(text));

  const finalRelease =
    /release\s+of\s+final\s+drawing(?:\(s\))?\s+for\s+production/i.test(text) ||
    (/final\s+(?:release\s+)?drawing(?:\(s\))?/i.test(text) &&
     /production/i.test(text));

  if (finalRelease) return 'FINAL';
  if (prelist) return 'PRELIST';
  return '';
}

function extractSnListFromRoutingEmail_(text) {
  const raw = String(text || '');
  const out = [];
  const seen = {};

  // Examples:
  // SN 44375-80, 44381, YSX40E
  // SN 42617-20, STS45
  // S/N 44256
  const re = /\bS\/?N\s*[:#-]?\s*([0-9][0-9,\-+\s;；，]*)/ig;
  let m;

  while ((m = re.exec(raw)) !== null) {
    let block = String(m[1] || '')
      .replace(/[，；]/g, function(ch) { return ch === '，' ? ',' : ';'; })
      .trim();

    // Stop overly long accidental captures.
    if (block.length > 100) block = block.substring(0, 100);

    const expanded = expandSnRange_(block);

    expanded.forEach(function(sn) {
      const key = normalizeSn_(sn);
      if (!key || seen[key]) return;
      seen[key] = true;
      out.push(String(sn));
    });
  }

  return out;
}

function extractFinalRoutingPartNumbers_(plainBody) {
  const body = String(plainBody || '');
  const lines = body.split(/\r?\n/);
  const out = [];
  const seen = {};

  lines.forEach(function(line) {
    const text = String(line || '');
    const upper = text.toUpperCase();

    // Exclude obvious reference/header lines that are not the E-part list.
    if (/\bGA\s*#/i.test(text) || /\bSO\s*#/i.test(text) || /\bECO\b/i.test(text)) {
      return;
    }

    // Typical Object ID format, e.g. BR1351955.
    const matches = text.match(/\b[A-Z]{1,4}\d{6,}\b/g) || [];

    matches.forEach(function(part) {
      const p = String(part || '').toUpperCase();

      // Defensive exclusions.
      if (/^SN\d+/i.test(p) || /^SO\d+/i.test(p) || /^ECO\d+/i.test(p)) return;

      if (!seen[p]) {
        seen[p] = true;
        out.push(p);
      }
    });
  });

  return out;
}


function xmlDecodeRouting_(value) {
  return String(value || '')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&quot;/g, '"')
    .replace(/&apos;/g, "'")
    .replace(/&amp;/g, '&');
}

function xlsxRowsFromAttachment_(att) {
  const rows = [];
  let files = [];
  try {
    files = Utilities.unzip(att.copyBlob ? att.copyBlob() : att);
  } catch (e) {
    return rows;
  }

  const byName = {};
  files.forEach(function(blob) {
    byName[String(blob.getName() || '')] = blob;
  });

  const shared = [];
  const ssBlob = byName['xl/sharedStrings.xml'];
  if (ssBlob) {
    const xml = String(ssBlob.getDataAsString('UTF-8') || '');
    const siRx = /<si\b[^>]*>([\s\S]*?)<\/si>/g;
    let sm;
    while ((sm = siRx.exec(xml)) !== null) {
      const parts = [];
      const tRx = /<t\b[^>]*>([\s\S]*?)<\/t>/g;
      let tm;
      while ((tm = tRx.exec(sm[1])) !== null) parts.push(xmlDecodeRouting_(tm[1]));
      shared.push(parts.join(''));
    }
  }

  Object.keys(byName)
    .filter(function(name) { return /^xl\/worksheets\/sheet\d+\.xml$/i.test(name); })
    .sort()
    .forEach(function(name) {
      const xml = String(byName[name].getDataAsString('UTF-8') || '');
      const rowRx = /<row\b[^>]*>([\s\S]*?)<\/row>/g;
      let rm;
      while ((rm = rowRx.exec(xml)) !== null) {
        const vals = [];
        const cellRx = /<c\b([^>]*)>([\s\S]*?)<\/c>/g;
        let cm;
        let nextColumn = 0;
        while ((cm = cellRx.exec(rm[1])) !== null) {
          const attrs = cm[1] || '';
          const inner = cm[2] || '';
          const refMatch = attrs.match(/\br="([A-Z]+)\d+"/i);
          let column = nextColumn;
          if (refMatch) {
            column = 0;
            String(refMatch[1]).toUpperCase().split('').forEach(function(ch) {
              column = column * 26 + (ch.charCodeAt(0) - 64);
            });
            column = Math.max(0, column - 1);
          }
          const typeMatch = attrs.match(/\bt="([^"]+)"/i);
          const type = typeMatch ? typeMatch[1] : '';
          let value = '';
          if (type === 'inlineStr') {
            const tm = inner.match(/<t\b[^>]*>([\s\S]*?)<\/t>/i);
            value = tm ? xmlDecodeRouting_(tm[1]) : '';
          } else {
            const vm = inner.match(/<v\b[^>]*>([\s\S]*?)<\/v>/i);
            value = vm ? xmlDecodeRouting_(vm[1]) : '';
            if (type === 's' && /^\d+$/.test(value)) value = shared[Number(value)] || '';
          }
          vals[column] = String(value || '').trim();
          nextColumn = column + 1;
        }
        if (vals.some(Boolean)) rows.push(vals);
      }
    });

  return rows;
}

function textRowsFromRoutingAttachment_(att) {
  const name = String(att.getName ? att.getName() : '').toLowerCase();
  if (/\.(xlsx|xlsm)$/i.test(name)) return xlsxRowsFromAttachment_(att);

  if (/\.(csv|txt|tsv|html?|xml)$/i.test(name)) {
    let text = '';
    try { text = String(att.getDataAsString('UTF-8') || ''); } catch (e) { text = ''; }
    if (/\.(html?|xml)$/i.test(name)) text = text.replace(/<[^>]+>/g, ' ').replace(/&nbsp;/gi, ' ');
    return text.split(/\r?\n/).map(function(line) {
      return line.split(/[\t,;]/).map(function(x) { return String(x || '').trim(); });
    }).filter(function(row) { return row.some(Boolean); });
  }

  return [];
}

function extractRoutingPartCandidates_(text) {
  const upper = String(text || '').toUpperCase();
  const out = [];
  const seen = {};
  const matches = upper.match(/\b(?:[A-Z]{1,5}\d{5,}|\d{6,})\b/g) || [];

  matches.forEach(function(value) {
    const v = String(value || '').trim();
    if (!v || /^SN\d+/i.test(v) || /^SO\d+/i.test(v) || /^ECO\d+/i.test(v)) return;
    if (!seen[v]) { seen[v] = true; out.push(v); }
  });
  return out;
}

function extractPrelistNoRoutingPartsFromAttachments_(attachments) {
  const out = [];
  const seen = {};
  function add_(value) {
    const part = String(value || '').toUpperCase();
    if (part && !seen[part]) { seen[part] = true; out.push(part); }
  }

  (attachments || []).forEach(function(att) {
    const rows = textRowsFromRoutingAttachment_(att);
    const name = String(att.getName ? att.getName() : '');
    let routingColumn = -1;
    let headerRow = -1;
    const fileIsNoRoutingList = /\bno[\s_-]*(?:routing|route)\b/i.test(name);
    let noRoutingColumn = -1;

    rows.forEach(function(row, rowIndex) {
      (row || []).forEach(function(cell, columnIndex) {
        const normalized = String(cell || '').replace(/[_-]+/g, ' ').replace(/\s+/g, ' ').trim();
        if (/^(?:routing|route)(?:\s+status)?$/i.test(normalized)) {
          routingColumn = columnIndex;
          headerRow = rowIndex;
        }
        if (/^no\s*(?:routing|route)$/i.test(normalized)) {
          noRoutingColumn = columnIndex;
        }
      });
    });

    rows.forEach(function(row, rowIndex) {
      const joined = (row || []).join(' | ');
      const ids = extractRoutingBrIdsLoose_(joined);
      if (!ids.length) return;
      const explicit = /\b(?:no\s*(?:routing|route)|not\s+routed|routing\s+(?:missing|blank|not\s+set))\b/i.test(joined);
      const compactDashRow = (row || []).filter(function(cell) { return String(cell || '').trim() !== ''; }).length <= 3 &&
        (row || []).some(function(cell) { return /^(?:-|–|—|n\/?a|none|blank|not\s+set)$/i.test(String(cell || '').trim()); });
      let missingByColumn = false;
      if (routingColumn >= 0 && rowIndex > headerRow) {
        const status = String((row || [])[routingColumn] || '').trim();
        missingByColumn = /^(?:|-|–|—|n\/?a|none|blank|no|not\s+set)$/i.test(status);
      }
      if (explicit || compactDashRow || missingByColumn || fileIsNoRoutingList) ids.forEach(add_);
      if (noRoutingColumn >= 0) extractRoutingBrIdsLoose_((row || [])[noRoutingColumn]).forEach(add_);
    });

    // Some senders attach a clean one/two-column BR list with no heading.
    // Accept it only when the attachment is overwhelmingly made of BR rows.
    const usefulRows = rows.filter(function(row) { return (row || []).some(function(cell) { return String(cell || '').trim(); }); });
    const idRows = usefulRows.filter(function(row) { return extractRoutingBrIdsLoose_((row || []).join(' | ')).length > 0; });
    if (routingColumn < 0 && noRoutingColumn < 0 && !fileIsNoRoutingList && idRows.length >= 2 && idRows.length / Math.max(1, usefulRows.length) >= 0.65) {
      idRows.forEach(function(row) { extractRoutingBrIdsLoose_((row || []).join(' | ')).forEach(add_); });
    }
  });

  return out;
}

function getRoutingOriginalEmail(threadId, messageId, imagesOnly) {
  const tid = String(threadId || '').trim();
  const mid = String(messageId || '').trim();
  const cache = CacheService.getUserCache();
  const key = (imagesOnly ? 'routing_original_v815_final_' : 'routing_original_v814_') + (mid || tid);

  const cached = cache.get(key);
  if (cached) {
    try { return JSON.parse(cached); } catch (e) {}
  }

  let msg = null;
  if (mid) {
    try { msg = GmailApp.getMessageById(mid); } catch (e) { msg = null; }
  }

  if (!msg && tid) {
    try {
      const thread = GmailApp.getThreadById(tid);
      const msgs = thread ? thread.getMessages() : [];
      msg = msgs.length ? msgs[msgs.length - 1] : null;
    } catch (e) { msg = null; }
  }

  if (!msg) throw new Error('Unable to load the selected Routing email from Gmail.');

  const thread = msg.getThread();
  let attachments = [];
  try {
    attachments = (msg.getAttachments({includeInlineImages:false, includeAttachments:true}) || []).map(function(a) {
      return {
        name: String(a.getName() || 'Attachment'),
        size: Number(a.getBytes ? a.getBytes().length : 0) || 0
      };
    });
  } catch (e) { attachments = []; }

  let body = '';
  try { body = String(msg.getPlainBody() || ''); } catch (e) { body = ''; }
  if (body.length > 60000) body = body.slice(0, 60000) + '\n\n[Message truncated in Workbench]';

  const imageParts = routingCollectMessageImages_(msg);
  // Retained for compatibility with an older deployed Index.html.
  const inlineImages = routingImagePayload_(imageParts);
  const htmlBody = routingSafeMessageHtml_(msg, imageParts, inlineImages);
  const fromHasmanizam = /hasmanizam/i.test(String(msg.getFrom() || ''));
  const ocrInfo = imagesOnly && !fromHasmanizam
    ? {text:'', evidence:[], imageCount:0, error:''}
    : routingOcrMessageImages_(msg, imageParts, !!imagesOnly);
  const finalBodyIds = imagesOnly && fromHasmanizam ? extractFinalHasmanizamObjectIds_(body) : [];

  const result = {
    ok: true,
    id: String(msg.getId() || ''),
    threadId: thread ? String(thread.getId() || '') : tid,
    permalink: thread ? String(thread.getPermalink() || '') : '',
    subject: String(msg.getSubject() || ''),
    from: String(msg.getFrom() || ''),
    to: String(msg.getTo() || ''),
    cc: String(msg.getCc() || ''),
    date: msg.getDate() ? formatDateTime_(msg.getDate()) : '',
    body: body,
    htmlBody: htmlBody,
    attachments: attachments,
    inlineImages: inlineImages,
    ocrIds: extractRoutingBrIdsLoose_(ocrInfo.text),
    finalBodyIds: finalBodyIds,
    ocrPrelistNoRoutingIds: ocrInfo.prelistNoRoutingIds || [],
    ocrEvidence: ocrInfo.evidence || [],
    ocrError: ocrInfo.error || ''
  };

  try { cache.put(key, JSON.stringify(result), 600); } catch (e) {}
  return result;
}

function getRoutingWorkState(sn) {
  sn = String(sn || '').trim();
  if (!sn) {
    return {
      sn: '',
      status: 'NOT STARTED',
      activeSessionId: '',
      startedAt: '',
      completedAt: '',
      lastActionAt: '',
      events: []
    };
  }

  const ss = getOrCreateWorkbenchStore_();
  const sh = ensureWorkbenchStore_(ss);
  const lastRow = sh.getLastRow();
  if (lastRow < 2) {
    return {
      sn: sn,
      status: 'NOT STARTED',
      activeSessionId: '',
      startedAt: '',
      completedAt: '',
      lastActionAt: '',
      events: []
    };
  }

  const values = sh.getRange(2, 1, lastRow - 1, 8).getValues();
  const norm = normalizeSn_(sn);

  const rows = values.filter(function(r) {
    return normalizeSn_(r[1]) === norm &&
           String(r[2] || '').toUpperCase() === 'ROUTING';
  });

  const sessions = {};
  rows.forEach(function(r) {
    const sessionId = String(r[4] || '').trim();
    if (!sessionId) return;
    if (!sessions[sessionId]) sessions[sessionId] = [];
    sessions[sessionId].push(r);
  });

  let latestStart = null;
  let latestSessionId = '';

  Object.keys(sessions).forEach(function(id) {
    sessions[id].forEach(function(r) {
      if (String(r[3] || '').toUpperCase() !== 'START') return;
      const ts = r[0] instanceof Date ? r[0] : new Date(r[0]);
      if (!latestStart || ts > latestStart) {
        latestStart = ts;
        latestSessionId = id;
      }
    });
  });

  let status = 'NOT STARTED';
  let startedAt = '';
  let completedAt = '';
  let activeSessionId = '';

  if (latestSessionId) {
    const sessionRows = sessions[latestSessionId] || [];
    const startRow = sessionRows.find(function(r) {
      return String(r[3] || '').toUpperCase() === 'START';
    });

    const completeRows = sessionRows.filter(function(r) {
      return String(r[3] || '').toUpperCase() === 'COMPLETE';
    });

    if (startRow) {
      startedAt = formatDateTime_(startRow[0] instanceof Date ? startRow[0] : new Date(startRow[0]));
    }

    if (completeRows.length) {
      const done = completeRows[completeRows.length - 1][0];
      completedAt = formatDateTime_(done instanceof Date ? done : new Date(done));
      status = 'COMPLETED';
    } else {
      status = 'IN PROGRESS';
      activeSessionId = latestSessionId;
    }
  }

  const events = rows
    .slice()
    .sort(function(a, b) {
      return new Date(b[0]).getTime() - new Date(a[0]).getTime();
    })
    .slice(0, 20)
    .map(function(r) {
      const ts = r[0] instanceof Date ? r[0] : new Date(r[0]);
      return {
        time: formatDateTime_(ts),
        action: String(r[3] || ''),
        sessionId: String(r[4] || ''),
        user: String(r[5] || ''),
        note: String(r[6] || '')
      };
    });

  return {
    sn: sn,
    status: status,
    activeSessionId: activeSessionId,
    startedAt: startedAt,
    completedAt: completedAt,
    lastActionAt: events.length ? events[0].time : '',
    events: events
  };
}

function chatMirrorColumns_(header) {
  const out = { received:null, from:null, space:null, conversationType:null, message:null, chatLink:null, messageId:null };
  (header || []).forEach(function(v, i) {
    const h = normalizeHeader_(v);
    if (h === 'received' || h === 'posttime' || h === 'time') out.received = i;
    else if (h === 'from' || h === 'sender' || h === 'sendersfullname') out.from = i;
    else if (h === 'space' || h === 'spacename') out.space = i;
    else if (h === 'conversationtype' || h === 'type') out.conversationType = i;
    else if (h === 'message' || h === 'messagetext') out.message = i;
    else if (h === 'chatlink' || h === 'linktothemesssageinchat' || h === 'linktothemessageinchat') out.chatLink = i;
    else if (h === 'messageid') out.messageId = i;
  });
  return out;
}

function chatMirrorCell_(row, idx) {
  return idx == null ? '' : String((row || [])[idx] == null ? '' : (row || [])[idx]).trim();
}

function chatMirrorPulseForSheet_(sh) {
  const lastRow = Math.max(0, Number(sh.getLastRow() || 0));
  const lastCol = Math.min(12, Math.max(1, Number(sh.getLastColumn() || 1)));
  let tail = '';
  if (lastRow >= 2) {
    const startRow = Math.max(2, lastRow - 2);
    const values = sh.getRange(startRow, 1, lastRow - startRow + 1, lastCol).getDisplayValues();
    tail = values.map(function(row) { return row.join('\u001f'); }).join('\u001e');
  }
  const raw = [lastRow, lastCol, tail].join('|');
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, raw, Utilities.Charset.UTF_8);
  return {
    pulse: Utilities.base64EncodeWebSafe(bytes).replace(/=+$/g, ''),
    lastRow: lastRow,
    checkedAt: formatDateTime_(new Date())
  };
}

function getChatMirrorPulse() {
  const settings = getSettings();
  const url = String(settings.chatMessagesUrl || '').trim();
  const tab = String(settings.chatMessagesTab || CHAT_MESSAGES_DEFAULT_TAB).trim() || CHAT_MESSAGES_DEFAULT_TAB;
  if (!url) return { configured:false, pulse:'', lastRow:0, checkedAt:formatDateTime_(new Date()) };
  const sh = openSheets_(url, tab, false)[0];
  const p = chatMirrorPulseForSheet_(sh);
  p.configured = true;
  return p;
}

function getChatMirror(forceRefresh) {
  const settings = getSettings();
  const url = String(settings.chatMessagesUrl || '').trim();
  const tab = String(settings.chatMessagesTab || CHAT_MESSAGES_DEFAULT_TAB).trim() || CHAT_MESSAGES_DEFAULT_TAB;
  const cache = CacheService.getUserCache();
  const cacheKey = 'chat_mirror_v720';

  if (!forceRefresh) {
    const cached = cache.get(cacheKey);
    if (cached) {
      try { return JSON.parse(cached); } catch (e) {}
    }
  }

  if (!url) {
    return {
      configured:false,
      messages:[],
      sheetName:'',
      spreadsheetName:'',
      loadedAt:formatDateTime_(new Date()),
      pulse:'',
      message:'Chat Messages Google Sheet is not configured.'
    };
  }

  const sh = openSheets_(url, tab, false)[0];
  const pulseInfo = chatMirrorPulseForSheet_(sh);
  const data = sh.getDataRange().getDisplayValues();
  if (!data.length) {
    return {configured:true,messages:[],sheetName:sh.getName(),spreadsheetName:sh.getParent().getName(),loadedAt:formatDateTime_(new Date()),pulse:pulseInfo.pulse,message:'Chat Messages sheet is empty.'};
  }

  const cols = chatMirrorColumns_(data[0]);
  if (cols.message == null || cols.from == null) {
    throw new Error('Chat Messages sheet must contain at least "From" and "Message" columns in row 1.');
  }

  const messages = [];
  for (let r=1; r<data.length; r++) {
    const row = data[r] || [];
    const message = chatMirrorCell_(row, cols.message);
    const from = chatMirrorCell_(row, cols.from);
    if (!message && !from) continue;
    const typeRaw = chatMirrorCell_(row, cols.conversationType).toUpperCase();
    const type = typeRaw.indexOf('DIRECT') >= 0 ? 'DIRECT' : (typeRaw || 'SPACE / GROUP');
    const space = chatMirrorCell_(row, cols.space);
    const messageId = chatMirrorCell_(row, cols.messageId);
    const conversationKey = (type === 'DIRECT' && !space)
      ? ('DIRECT|' + from.toUpperCase())
      : (type + '|' + (space || from || 'Chat').toUpperCase());
    messages.push({
      rowNumber:r+1,
      received:chatMirrorCell_(row, cols.received),
      from:from,
      space:space,
      conversationType:type,
      message:message,
      chatLink:chatMirrorCell_(row, cols.chatLink),
      messageId:messageId,
      conversationKey:conversationKey
    });
  }

  const result = {
    configured:true,
    messages:messages,
    sheetName:sh.getName(),
    spreadsheetName:sh.getParent().getName(),
    loadedAt:formatDateTime_(new Date()),
    pulse:pulseInfo.pulse,
    message:''
  };
  try { cache.put(cacheKey, JSON.stringify(result), 30); } catch (e) {}
  return result;
}

function ensureComponentLog_(ss) {
  let sh = ss.getSheetByName(WORKBENCH_COMPONENT_LOG_SHEET);
  if (!sh) sh = ss.insertSheet(WORKBENCH_COMPONENT_LOG_SHEET);

  const headers = [
    'Group Key',
    'S/N Group',
    'S/N List',
    'Model',
    'Production Date',
    'Allocation Due',
    'Status',
    'Completed At',
    'User',
    'Tab Name',
    'Tab GID',
    'GA',
    'Scheduled S/N Snapshot',

  ];

  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);

function componentTabName_(label, sns) {
  let name = String(label || '').trim();
  const list = Array.isArray(sns) ? sns.filter(Boolean) : [];
  if (!name) name = list.length ? list.join(', ') : 'Component Allocation';
  name = name.replace(/[:\\\/\?\*\[\]]/g, ' ').replace(/\s+/g, ' ').trim();
  if (name.length > 96) {
    const first = list.length ? String(list[0]) : name.substring(0, 72);
    name = first + (list.length > 1 ? (' +' + (list.length - 1) + ' SN') : '');
  }
  if (!name) name = 'Component Allocation';
  return name.substring(0, 99);
}

function componentSnListFromCell_(value) {
  return String(value || '')
    .split(',')
    .map(function(x){ return normalizeSnDisplay_(x.trim()); })
    .filter(Boolean);
}

function componentSnFingerprint_(list) {
  return (Array.isArray(list) ? list : [])
    .map(function(x){ return normalizeSn_(x); })
    .filter(Boolean)
    .sort()
    .join(',');
}

function componentShortDate_(value) {
  if (value instanceof Date && !isNaN(value.getTime())) {
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'dd-MMM');
  }
  const s = String(value || '').trim();
  if (!s) return '';
  // Google Sheets may have previously serialized a date-like cell into a long
  // JavaScript date string. Normalize it for the DONE view without changing
  // the underlying log data.
  if (/GMT[+-]\d{4}|\b(?:Mon|Tue|Wed|Thu|Fri|Sat|Sun)\b/.test(s)) {
    const d = new Date(s);
    if (!isNaN(d.getTime())) return Utilities.formatDate(d, Session.getScriptTimeZone(), 'dd-MMM');
  }
  return s;
}

function componentCompletedAtShort_(value) {
  if (value instanceof Date && !isNaN(value.getTime())) {
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'dd-MMM-yyyy HH:mm');
  }
  const s = String(value || '').trim();
  if (!s) return '';
  const d = new Date(s);
  if (!isNaN(d.getTime())) return Utilities.formatDate(d, Session.getScriptTimeZone(), 'dd-MMM-yyyy HH:mm');
  return s;
}

function getComponentAllocationState() {
  const settings = getSettings();
  const store = getOrCreateWorkbenchStore_();
  const sh = ensureComponentLog_(store);
  const lastRow = sh.getLastRow();
  const done = {};
  const doneItems = [];

  if (lastRow >= 2) {
    const values = sh.getRange(2, 1, lastRow - 1, 14).getValues();
    values.forEach(function(r, idx) {
      const key = String(r[0] || '').trim();
      if (!key) return;
      const status = String(r[6] || '').trim().toUpperCase();
      if (status !== 'DONE') return;
      const groupSns = componentSnListFromCell_(r[2]);
      const scheduledSnapshot = componentSnListFromCell_(r[12]);
      const item = {
        rowNumber: idx + 2,
        groupKey: key,
        label: String(r[1] || ''),
        sns: groupSns,
        scheduledSns: scheduledSnapshot.length ? scheduledSnapshot : groupSns,
        model: String(r[3] || ''),
        productionDate: componentShortDate_(r[4]),
        dueDate: componentShortDate_(r[5]),
        status: 'DONE',
        completedAt: componentCompletedAtShort_(r[7]),
        completedAtMs: r[7] instanceof Date ? r[7].getTime() : 0,
        user: String(r[8] || ''),
        tabName: String(r[9] || ''),
        tabGid: String(r[10] || ''),
        ga: String(r[11] || ''),
        gaType: String(r[13] || 'NEW GA').trim().toUpperCase() || 'NEW GA'
      };
      doneItems.push(item);
      const previous = done[key];
      if (!previous || Number(item.completedAtMs || 0) >= Number(previous.completedAtMs || 0)) {
        done[key] = item;
      }
    });
  }

  doneItems.sort(function(a,b){
    const am = Number(a.completedAtMs || 0), bm = Number(b.completedAtMs || 0);
    if (am !== bm) return bm - am;
    return Number(b.rowNumber || 0) - Number(a.rowNumber || 0);
  });

  let spreadsheetName = '';
  let templateFound = false;
  if (settings.componentAllocationUrl) {
    try {
      const target = SpreadsheetApp.openByUrl(settings.componentAllocationUrl);
      spreadsheetName = target.getName();
      templateFound = !!target.getSheetByName(String(settings.componentTemplateTab || 'TEMPLATE').trim() || 'TEMPLATE');
    } catch (e) {
      spreadsheetName = '';
    }
  }

  return {
    configured: !!settings.componentAllocationUrl,
    spreadsheetName: spreadsheetName,
    templateTab: String(settings.componentTemplateTab || 'TEMPLATE').trim() || 'TEMPLATE',
    templateFound: templateFound,
    done: done,
    doneItems: doneItems,
    loadedAt: formatDateTime_(new Date())
  };
}

function openComponentAllocationTab(group) {
  group = group || {};
  const settings = getSettings();
  if (!settings.componentAllocationUrl) {
    throw new Error('Please bind the Component Allocation Google Sheet in Settings first.');
  }

  const ss = SpreadsheetApp.openByUrl(settings.componentAllocationUrl);
  const sns = Array.isArray(group.sns) ? group.sns : [];
  const tabName = componentTabName_(group.label, sns);
  let sh = null;
  let created = false;

  // DONE history stores the exact allocation tab. Prefer that stable tab reference
  // so clicking a completed S/N returns to the same Google Sheet tab.
  const savedGid = Number(group.tabGid || 0);
  if (savedGid) {
    sh = ss.getSheets().find(function(s){ return Number(s.getSheetId()) === savedGid; }) || null;
  }
  if (!sh && group.tabName) {
    sh = ss.getSheetByName(String(group.tabName));
  }
  if (!sh) {
    sh = ss.getSheetByName(tabName);
  }

  if (!sh) {
    const templateName = String(settings.componentTemplateTab || 'TEMPLATE').trim() || 'TEMPLATE';
    const template = ss.getSheetByName(templateName);
    if (template) {
      sh = template.copyTo(ss);
      sh.setName(tabName);
    } else {
      sh = ss.insertSheet(tabName);
    }
    created = true;
  }

  return {
    ok: true,
    created: created,
    spreadsheetName: ss.getName(),
    tabName: sh.getName(),
    gid: sh.getSheetId(),
    url: ss.getUrl() + '#gid=' + sh.getSheetId()
  };
}

function completeComponentAllocation(group) {
  group = group || {};
  const key = String(group.groupKey || '').trim();
  if (!key) throw new Error('Component design-group key is missing. Refresh and try again.');

  const ss = getOrCreateWorkbenchStore_();
  const sh = ensureComponentLog_(ss);
  const settings = getSettings();
  let tabName = componentTabName_(group.label, group.sns || []);
  let tabGid = '';
  if (settings.componentAllocationUrl) {
    try {
      const target = SpreadsheetApp.openByUrl(settings.componentAllocationUrl);
      const tab = target.getSheetByName(tabName);
      if (tab) tabGid = String(tab.getSheetId());
    } catch (e) {}
  }

  const scheduledSns = (Array.isArray(group.scheduledSns) && group.scheduledSns.length)
    ? group.scheduledSns
    : (Array.isArray(group.sns) ? group.sns : []);
  const snapshotText = scheduledSns.join(', ');
  const snapshotFingerprint = componentSnFingerprint_(scheduledSns);
  let existingRow = 0;
  const lastRow = sh.getLastRow();

  // Idempotency: completing the exact same design-group + scheduled S/N snapshot
  // updates that completion instead of creating a duplicate history row.
  if (lastRow >= 2 && snapshotFingerprint) {
    const existing = sh.getRange(2, 1, lastRow - 1, 14).getValues();
    for (let i = existing.length - 1; i >= 0; i--) {
      if (String(existing[i][0] || '').trim() !== key) continue;
      if (String(existing[i][6] || '').trim().toUpperCase() !== 'DONE') continue;
      const oldSnapshot = componentSnListFromCell_(existing[i][12]);
      const fallback = oldSnapshot.length ? oldSnapshot : componentSnListFromCell_(existing[i][2]);
      if (componentSnFingerprint_(fallback) === snapshotFingerprint) {
        existingRow = i + 2;
        break;
      }
    }
  }

  const row = [[
    key,
    String(group.label || ''),
    (Array.isArray(group.sns) ? group.sns : []).join(', '),
    String(group.model || ''),
    String(group.productionDate || ''),
    String(group.dueDate || ''),
    'DONE',
    new Date(),
    workbenchUser_(),
    tabName,
    tabGid,
    String(group.ga || ''),
    snapshotText,
    String(group.gaType || 'NEW GA').trim().toUpperCase() || 'NEW GA'
  ]];

  if (existingRow) sh.getRange(existingRow, 1, 1, 14).setValues(row);
  else sh.getRange(sh.getLastRow() + 1, 1, 1, 14).setValues(row);

  return getComponentAllocationState();
}

function getOrCreateHistoryStore_() {
  const props = PropertiesService.getScriptProperties();
  const existingId = props.getProperty(HISTORY_FILE_ID_KEY);

  if (existingId) {
    try {
      const ss = SpreadsheetApp.openById(existingId);
      ensureHistoryStore_(ss);
      return ss;
    } catch (e) {
      // If the file was deleted or access changed, create a new one.
    }
  }

  const ss = SpreadsheetApp.create(APP_NAME + ' - History');
  props.setProperty(HISTORY_FILE_ID_KEY, ss.getId());

  const first = ss.getSheets()[0];
  first.setName(HISTORY_SNAPSHOT_SHEET);

  const log = ss.insertSheet(HISTORY_LOG_SHEET);
  ensureHistoryStore_(ss);

  return ss;
}

  try {
    if (!snapshot.isSheetHidden()) snapshot.hideSheet();
  } catch (e) {}
}

function buildPersistentSnapshot_(rows, productionRows) {
  const map = {};

  (rows || []).forEach(function(r) {
    const sn = normalizeSnDisplay_(r.sn);
    if (!sn) return;

    map['M|' + normalizeSn_(sn)] = {
      key: 'M|' + normalizeSn_(sn),
      source: 'MASTERPLAN',
      sn: sn,
      data: {
        seq: r.seq || '',
        sapSo: r.sapSo || '',
        model: r.model || '',
        customer: r.customer || '',
        endUser: r.endUser || '',
        weekKey: r.weekKey || '',
        comDate: r.comDate || '',

        engineerMatched: !!r.engineerMatched,
        engineerFallbackSource: r.engineerFallbackSource || '',
        mechEngineer: r.mechEngineer || '',
        elecEngineer: r.elecEngineer || '',
        mechanical: compactStatusSnapshot_(r.mechanical),
        electrical: compactStatusSnapshot_(r.electrical),

        // FTZ History rule V5.28:
        // only the actual Remarks cell text is persisted for comparison.
        routingRemarks: r.routingRemarks || ''
      }
    };
  });

  // A production S/N can occasionally appear more than once in a calendar.
  // Keep all placements for that S/N in a stable sorted array, so moving the
  // S/N to another date/week is detected as one production schedule change.
  const prodBySn = {};
  (productionRows || []).forEach(function(r) {
    const sn = normalizeSnDisplay_(r.sn);
    if (!sn) return;
    const k = normalizeSn_(sn);
    if (!prodBySn[k]) prodBySn[k] = {
      sn: sn,
      placements: []
    };
    prodBySn[k].placements.push({
      productionDate: r.productionDate || '',
      productionDateKey: r.productionDateKey || '',
      productionWeekKey: r.productionWeekKey || '',
      model: r.model || '',
      productionStatus: r.productionStatus || '',
      sourceTab: r.sourceTab || ''
    });
  });

  Object.keys(prodBySn).forEach(function(k) {
    const item = prodBySn[k];
    item.placements.sort(function(a, b) {
      return JSON.stringify(a).localeCompare(JSON.stringify(b));
    });
    map['P|' + k] = {
      key: 'P|' + k,
      source: 'PRODUCTION',
      sn: item.sn,
      data: {
        placements: item.placements
      }
    };
  });

  return map;
}

function readPersistentSnapshot_(sheet) {
  const out = {};
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return out;

  const values = sheet.getRange(2, 1, lastRow - 1, 4).getValues();
  values.forEach(function(row) {
    const key = String(row[0] || '').trim();
    if (!key) return;

    let data = {};
    try {
      data = JSON.parse(String(row[3] || '{}'));
    } catch (e) {}

    out[key] = {
      key: key,
      source: String(row[1] || ''),
      sn: String(row[2] || ''),
      data: data
    };
  });

  return out;
}

function writePersistentSnapshot_(sheet, map) {
  const keys = Object.keys(map).sort();

  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, 4).clearContent();
  }

  if (!keys.length) return;

  const values = keys.map(function(key) {
    const item = map[key];
    return [
      item.key,
      item.source,
      item.sn,
      JSON.stringify(item.data || {})
    ];
  });

  sheet.getRange(2, 1, values.length, 4).setValues(values);
}

function stringifyHistoryValue_(v) {
  if (v === null || v === undefined || v === '') return '-';
  if (Array.isArray(v) || typeof v === 'object') return JSON.stringify(v);
  return String(v);
}

function productionDateHistoryText_(placements) {
  if (!placements || !placements.length) return '-';

  // Production History is date-only.
  // Ignore Model, Status, source tab and any other schedule details.
  const byDate = {};

  (placements || []).forEach(function(p) {
    const key = String(p.productionDateKey || p.productionDate || '').trim();
    if (!key) return;

    if (!byDate[key]) {
      byDate[key] = {
        key: key,
        display: String(p.productionDate || p.productionDateKey || '').trim()
      };
    }
  });

  const items = Object.keys(byDate).map(function(k) {
    return byDate[k];
  });

  items.sort(function(a, b) {
    return String(a.key).localeCompare(String(b.key));
  });

  return items.length
    ? items.map(function(x) { return x.display || x.key; }).join(' ; ')
    : '-';
}

  ]);

    ['engineerFallbackSource', 'Engineer Fallback Source', 'ENGINEER'],
    ['mechEngineer', 'Mechanical Engineer', 'ENGINEER'],
    ['elecEngineer', 'Electrical Engineer', 'ENGINEER']
  ];

  simple.forEach(function(item) {
    const key = item[0];
    const label = item[1];
    const source = item[2];

    if (String(p[key] ?? '') !== String(c[key] ?? '')) {
      appendHistoryChange_(
        changes,
        source,
        sn,
        label,
        p[key],
        c[key],
        label + ': ' + stringifyHistoryValue_(p[key]) + ' → ' + stringifyHistoryValue_(c[key])
      );
    }
  });

  // V5.28: FTZ Schedule history records ONLY the Remarks cell.
  // Derived Routing status/date/latest-text changes are intentionally ignored.
  if (String(p.routingRemarks ?? '') !== String(c.routingRemarks ?? '')) {
    appendHistoryChange_(
      changes,
      'FTZ',
      sn,
      'Routing Remarks',
      p.routingRemarks,
      c.routingRemarks,
      'Routing Remarks: ' +
        stringifyHistoryValue_(p.routingRemarks) + ' → ' +
        stringifyHistoryValue_(c.routingRemarks)
    );
  }

  const statusGroups = [
    ['mechanical', 'Mechanical'],
    ['electrical', 'Electrical']
  ];

  const statusFields = [
    ['stage', 'Stage'],
    ['listPlan', 'List Plan'],
    ['listActual', 'List Actual'],
    ['arfPlan', 'ARF Plan'],
    ['arfActual', 'ARF Actual'],
    ['finalPlan', 'Final Plan'],
    ['finalActual', 'Final Actual'],
    ['overdueStep', 'Overdue']
  ];

  statusGroups.forEach(function(group) {
    const key = group[0];
    const label = group[1];
    const ps = p[key] || {};
    const cs = c[key] || {};

    statusFields.forEach(function(sf) {
      const f = sf[0];
      const flabel = label + ' ' + sf[1];

      if (String(ps[f] ?? '') !== String(cs[f] ?? '')) {
        appendHistoryChange_(
          changes,
          'ENGINEER',
          sn,
          flabel,
          ps[f],
          cs[f],
          flabel + ': ' + stringifyHistoryValue_(ps[f]) + ' → ' + stringifyHistoryValue_(cs[f])
        );
      }
    });
  });
}

function compareProductionSnapshot_(prev, cur, changes) {
  const sn = cur.sn || prev.sn || '';

  // V5.32:
  // Only the order's initial fabrication DATE is relevant to Production
  // Planner Change History. Status / Model / source-tab edits are ignored.
  const oldText = productionDateHistoryText_(
    (prev.data || {}).placements || []
  );
  const newText = productionDateHistoryText_(
    (cur.data || {}).placements || []
  );

  if (oldText !== newText) {
    appendHistoryChange_(
      changes,
      'PRODUCTION',
      sn,
      'Initial Fabrication Date',
      oldText,
      newText,
      'Initial fabrication date: ' + oldText + ' → ' + newText
    );
  }
}


function historyAffectedSnCount_(changes) {
  const seen = {};
  (changes || []).forEach(function(row) {
    const sn = String(row[2] || '').trim();
    if (sn) seen[sn] = true;
  });
  return Object.keys(seen).length;
}

function shouldMassChangeRebase_(changes, currentMap, previousMap) {
  const currentCount = Object.keys(currentMap || {}).length;
  const previousCount = Object.keys(previousMap || {}).length;
  const baseCount = Math.max(currentCount, previousCount, 1);

  const affectedSn = historyAffectedSnCount_(changes);
  const threshold = Math.max(
    HISTORY_MASS_CHANGE_MIN_SN,
    Math.ceil(baseCount * HISTORY_MASS_CHANGE_RATIO)
  );

  return affectedSn >= threshold;
}


function comparePlannerMigration_(previousMap, currentMap) {
  const changes = [];
  const matched = [];
  const newSn = [];
  const missingSn = [];

  const fields = [
    ['sapSo', 'SAP SO'],
    ['model', 'Model'],
    ['customer', 'Customer'],
    ['endUser', 'End User'],
    ['weekKey', 'Week'],
    ['comDate', 'Com. Date']
  ];

  const currentMasterKeys = Object.keys(currentMap || {}).filter(function(k) {
    return k.indexOf('M|') === 0;
  });

  const previousMasterKeys = Object.keys(previousMap || {}).filter(function(k) {
    return k.indexOf('M|') === 0;
  });

  currentMasterKeys.forEach(function(key) {
    const cur = currentMap[key];
    const prev = previousMap[key];

    if (!prev || prev.source !== 'MASTERPLAN') {
      newSn.push(cur.sn || key.substring(2));
      return;
    }

    matched.push(cur.sn || key.substring(2));

    const p = prev.data || {};
    const c = cur.data || {};

    fields.forEach(function(item) {
      const fieldKey = item[0];
      const label = item[1];
      const before = String(p[fieldKey] ?? '').trim();
      const after = String(c[fieldKey] ?? '').trim();

      if (before === after) return;

      appendHistoryChange_(
        changes,
        'MASTERPLAN',
        cur.sn || prev.sn || '',
        label,
        before,
        after,
        'Recovered during Masterplan migration: ' +
          label + ': ' + stringifyHistoryValue_(before) +
          ' → ' + stringifyHistoryValue_(after)
      );
    });
  });

  previousMasterKeys.forEach(function(key) {
    if (currentMap[key] && currentMap[key].source === 'MASTERPLAN') return;
    const prev = previousMap[key];
    missingSn.push(prev.sn || key.substring(2));
  });

  newSn.sort();
  missingSn.sort();

  return {
    changes: changes,
    report: {
      performed: true,
      matchedSn: matched.length,
      recoveredChanges: changes.length,
      newSnCount: newSn.length,
      missingSnCount: missingSn.length,
      newSn: newSn.slice(0, 100),
      missingSn: missingSn.slice(0, 100),
      note:
        'Shared S/N field changes were recovered automatically. ' +
        'New / missing S/N are review-only and were not auto-written as Added / Removed.'
    }
  };
}

function resetChangeHistoryBaseline(clearHistory) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const ss = getOrCreateHistoryStore_();
    const snapshotSheet = ss.getSheetByName(HISTORY_SNAPSHOT_SHEET);

    // V5.4 safety rule:
    // Change History is permanent from the app UI. Even if an older frontend
    // accidentally calls this function with true, the log will NOT be cleared.
    if (snapshotSheet.getLastRow() > 1) {
      snapshotSheet
        .getRange(2, 1, snapshotSheet.getLastRow() - 1, 4)
        .clearContent();
    }

    const props = PropertiesService.getScriptProperties();
    props.setProperty(HISTORY_FORCE_REBASE_KEY, '1');
    props.deleteProperty(HISTORY_SOURCE_SIGNATURE_KEY);
    props.deleteProperty(HISTORY_PLANNER_MIGRATION_KEY);

    return {
      ok: true,
      clearedHistory: false,
      message: 'Baseline reset requested. Existing Change History is kept.'
    };
  } finally {
    lock.releaseLock();
  }
}

function syncPersistentHistory_(rows, productionRows, sourceSignature) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const ss = getOrCreateHistoryStore_();
    const snapshotSheet = ss.getSheetByName(HISTORY_SNAPSHOT_SHEET);
    const logSheet = ss.getSheetByName(HISTORY_LOG_SHEET);

    const props = PropertiesService.getScriptProperties();
    const previousSignature = props.getProperty(HISTORY_SOURCE_SIGNATURE_KEY) || '';
    const forceRebase = props.getProperty(HISTORY_FORCE_REBASE_KEY) === '1';
    const plannerMigration =
      props.getProperty(HISTORY_PLANNER_MIGRATION_KEY) === '1';

    const previous = readPersistentSnapshot_(snapshotSheet);
    const current = buildPersistentSnapshot_(rows, productionRows);

    // V5.28 explicit one-time full History reset.
    // Clear all stored Change History rows, then rebuild Snapshot from the
    // current Google Sheets so future changes start from a clean baseline.
    const lastResetVersion = props.getProperty(HISTORY_RESET_VERSION_KEY) || '';
    if (lastResetVersion !== HISTORY_RESET_VERSION) {
      if (logSheet.getLastRow() > 1) {
        logSheet
          .getRange(2, 1, logSheet.getLastRow() - 1, 7)
          .clearContent();
      }

      if (snapshotSheet.getLastRow() > 1) {
        snapshotSheet
          .getRange(2, 1, snapshotSheet.getLastRow() - 1, 4)
          .clearContent();
      }

      writePersistentSnapshot_(snapshotSheet, current);

      if (sourceSignature) {
        props.setProperty(HISTORY_SOURCE_SIGNATURE_KEY, sourceSignature);
      }

      props.setProperty(HISTORY_RESET_VERSION_KEY, HISTORY_RESET_VERSION);
      props.deleteProperty(HISTORY_FORCE_REBASE_KEY);
      props.deleteProperty(HISTORY_PLANNER_MIGRATION_KEY);

      const resetPage = readChangeHistoryPage_(logSheet, 0, HISTORY_PAGE_SIZE);
      resetPage.historyReset = true;
      resetPage.rebased = false;
      resetPage.rebaseReason = '';
      return resetPage;
    }

    const hadBaseline = Object.keys(previous).length > 0;
    let rebaseReason = '';

    // V5.5 Masterplan Migration Compare:
    // Preserve old history, compare old/new Masterplan by the same S/N,
    // recover only high-confidence field differences, then adopt the new
    // workbook as the baseline.
    if (plannerMigration && hadBaseline && !forceRebase) {
      const migration = comparePlannerMigration_(previous, current);

      if (migration.changes.length) {
        const migrationStartRow = logSheet.getLastRow() + 1;
        logSheet
          .getRange(migrationStartRow, 1, migration.changes.length, 7)
          .setValues(migration.changes);
      }

      writePersistentSnapshot_(snapshotSheet, current);

      if (sourceSignature) {
        props.setProperty(HISTORY_SOURCE_SIGNATURE_KEY, sourceSignature);
      }

      props.deleteProperty(HISTORY_PLANNER_MIGRATION_KEY);
      props.deleteProperty(HISTORY_FORCE_REBASE_KEY);

      const migrationPage =
        readChangeHistoryPage_(logSheet, 0, HISTORY_PAGE_SIZE);

      migrationPage.rebased = true;
      migrationPage.rebaseReason = 'Masterplan migration compared safely';
      migrationPage.migrationReport = migration.report;

      return migrationPage;
    }

    // Safety rule 1:
    // Changing Google Sheet URL, Tab, Read entire spreadsheet, or the actual
    // set of tabs being read creates a new baseline. It must NOT become history.
    if (forceRebase) {
      rebaseReason = 'Reference source settings changed';
    } else if (previousSignature && sourceSignature &&
               previousSignature !== sourceSignature) {
      rebaseReason = 'Reference workbook/tab structure changed';
    }

    // First baseline or source/config reset: replace snapshot only.
    if (!hadBaseline || rebaseReason) {
      writePersistentSnapshot_(snapshotSheet, current);

      if (sourceSignature) {
        props.setProperty(HISTORY_SOURCE_SIGNATURE_KEY, sourceSignature);
      }
      props.deleteProperty(HISTORY_FORCE_REBASE_KEY);
      props.deleteProperty(HISTORY_PLANNER_MIGRATION_KEY);

      const page = readChangeHistoryPage_(logSheet, 0, HISTORY_PAGE_SIZE);
      page.rebased = !!rebaseReason;
      page.rebaseReason = rebaseReason;
      return page;
    }

    const changes = [];

    Object.keys(current).forEach(function(key) {
      const cur = current[key];
      const prev = previous[key];

      if (!prev) {
        appendHistoryChange_(
          changes,
          cur.source,
          cur.sn,
          cur.source === 'PRODUCTION' ? 'Production Order' : 'Order',
          '-',
          'Added',
          cur.source === 'PRODUCTION'
            ? 'S/N added to Production Planner'
            : 'S/N added to Masterplan'
        );
        return;
      }

      if (cur.source === 'MASTERPLAN') {
        compareMasterSnapshot_(prev, cur, changes);
      } else if (cur.source === 'PRODUCTION') {
        compareProductionSnapshot_(prev, cur, changes);
      }
    });

    Object.keys(previous).forEach(function(key) {
      if (current[key]) return;

      const prev = previous[key];
      appendHistoryChange_(
        changes,
        prev.source,
        prev.sn,
        prev.source === 'PRODUCTION' ? 'Production Order' : 'Order',
        'Present',
        'Removed',
        prev.source === 'PRODUCTION'
          ? 'S/N removed from Production Planner'
          : 'S/N removed from Masterplan'
      );
    });

    // Safety rule 2:
    // If one refresh suddenly affects a very large portion of all S/N,
    // assume a workbook replacement/import/layout change and rebase instead
    // of flooding Change History.
    if (shouldMassChangeRebase_(changes, current, previous)) {
      writePersistentSnapshot_(snapshotSheet, current);

      if (sourceSignature) {
        props.setProperty(HISTORY_SOURCE_SIGNATURE_KEY, sourceSignature);
      }
      props.deleteProperty(HISTORY_FORCE_REBASE_KEY);
      props.deleteProperty(HISTORY_PLANNER_MIGRATION_KEY);

      const page = readChangeHistoryPage_(logSheet, 0, HISTORY_PAGE_SIZE);
      page.rebased = true;
      page.rebaseReason = 'Mass-change protection triggered';
      return page;
    }

    if (changes.length) {
      const startRow = logSheet.getLastRow() + 1;
      logSheet.getRange(startRow, 1, changes.length, 7).setValues(changes);
    }

    writePersistentSnapshot_(snapshotSheet, current);

    if (sourceSignature) {
      props.setProperty(HISTORY_SOURCE_SIGNATURE_KEY, sourceSignature);
    }
    props.deleteProperty(HISTORY_FORCE_REBASE_KEY);
    props.deleteProperty(HISTORY_PLANNER_MIGRATION_KEY);

    const page = readChangeHistoryPage_(logSheet, 0, HISTORY_PAGE_SIZE);
    page.rebased = false;
    page.rebaseReason = '';
    return page;

  } finally {
    lock.releaseLock();
  }
}

function readChangeHistoryPage_(sheet, offset, limit) {
  offset = Math.max(0, Number(offset || 0));
  limit = Math.max(1, Math.min(1000, Number(limit || HISTORY_PAGE_SIZE)));

  const total = Math.max(0, sheet.getLastRow() - 1);
  if (!total) return { items: [], total: 0, hasMore: false };

  const endIndex = total - offset;
  if (endIndex <= 0) return { items: [], total: total, hasMore: false };

  const count = Math.min(limit, endIndex);
  const startRow = 2 + (endIndex - count);

  const values = sheet.getRange(startRow, 1, count, 7).getValues();
  const items = values.reverse().map(function(row) {
    const ts = row[0] instanceof Date ? formatDateTime_(row[0]) : String(row[0] || '');
    return {
      time: ts,
      timestampMs: row[0] instanceof Date ? row[0].getTime() : 0,
      source: String(row[1] || ''),
      sn: String(row[2] || ''),
      field: String(row[3] || ''),
      previous: String(row[4] || ''),
      updated: String(row[5] || ''),
      desc: String(row[6] || '')
    };
  });

  return {
    items: items,
    total: total,
    hasMore: offset + count < total
  };
}

function getChangeHistoryPage(offset, limit) {
  const ss = getOrCreateHistoryStore_();
  const logSheet = ss.getSheetByName(HISTORY_LOG_SHEET);
  return readChangeHistoryPage_(logSheet, offset, limit);
}

function openSheets_(urlOrId, tabName, readAllTabs) {
  const text = String(urlOrId || '').trim();
  let ss;
  if (/^https?:\/\//i.test(text)) {
    ss = SpreadsheetApp.openByUrl(text);
  } else {
    ss = SpreadsheetApp.openById(text);
  }

  if (readAllTabs) {
    const sheets = ss.getSheets();
    if (!sheets.length) throw new Error('Google Sheet 内没有可读取的工作表。');
    return sheets;
  }

  if (tabName) {
    const sh = ss.getSheetByName(tabName);
    if (!sh) {
      throw new Error('找不到 Sheet tab: "' + tabName + '"，文件：' + ss.getName());
    }
    return [sh];
  }

  const sheets = ss.getSheets();
  if (!sheets.length) throw new Error('Google Sheet 内没有可读取的工作表。');
  return [sheets[0]];
}

function readSheet_(sheet) {
  const range = sheet.getDataRange();
  return {
    display: range.getDisplayValues(),
    raw: range.getValues()
  };
}


function parsePlannerSheets_(sheets) {
  let allRows = [];
  let warnings = [];
  sheets.forEach(function(sheet) {
    try {
      const parsed = parsePlanner_(readSheet_(sheet));
      parsed.rows.forEach(function(r) {
        r.sourceTab = sheet.getName();
      });
      allRows = allRows.concat(parsed.rows);
      warnings = warnings.concat((parsed.warnings || []).map(w => sheet.getName() + ': ' + w));
    } catch (e) {
      warnings.push(sheet.getName() + ': skipped - ' + e.message);
    }
  });

  // 重新生成跨 tab 的 Planner sequence。
  allRows.forEach(function(r, i) { r.seq = i + 1; });

  return { rows: allRows, warnings: warnings };
}

function parseEngineerSheets_(sheets) {
  const bySn = {};
  let warnings = [];

  sheets.forEach(function(sheet) {
    try {
      const parsed = parseEngineer_(readSheet_(sheet));
      Object.keys(parsed.bySn).forEach(function(key) {
        if (bySn[key]) {
          warnings.push('Duplicate Engineer S/N across tabs: ' + key + ' (' + bySn[key].sourceTab + ' / ' + sheet.getName() + ')');
        }
        parsed.bySn[key].sourceTab = sheet.getName();
        bySn[key] = parsed.bySn[key];
      });
      warnings = warnings.concat((parsed.warnings || []).map(w => sheet.getName() + ': ' + w));
    } catch (e) {
      warnings.push(sheet.getName() + ': skipped - ' + e.message);
    }
  });

  return { bySn: bySn, warnings: warnings };
}



function parseFtzSheets_(sheets) {
  const bySn = {};
  let warnings = [];

  sheets.forEach(function(sheet) {
    try {
      const parsed = parseFtzSchedule_(readSheet_(sheet), sheet.getName());

      Object.keys(parsed.bySn).forEach(function(key) {
        const incoming = parsed.bySn[key];
        const existing = bySn[key];

        if (!existing) {
          bySn[key] = incoming;
          return;
        }

        // V7.13: GA comes from FTZ Schedule column V. Preserve a non-blank GA
        // even when the routing-history winner is a different duplicate row/tab.
        if (isNewerFtzRecord_(incoming, existing)) {
          if (!incoming.ga && existing.ga) incoming.ga = existing.ga;
          bySn[key] = incoming;
        } else if (!existing.ga && incoming.ga) {
          existing.ga = incoming.ga;
        }
      });

      warnings = warnings.concat(
        (parsed.warnings || []).map(function(w) {
          return 'FTZ ' + sheet.getName() + ': ' + w;
        })
      );
    } catch (e) {
      warnings.push('FTZ ' + sheet.getName() + ': skipped - ' + e.message);
    }
  });

  return { bySn: bySn, warnings: warnings };
}

function parseFtzSchedule_(data, sheetName) {
  const values = data.display || [];
  const cols = detectColumns_(values, {
    sn: ['SN', 'S/N', 'Serial No', 'Serial Number'],
    remarks: ['Remarks', 'Remark']
  }, 20);

  if (cols.sn == null || cols.remarks == null) {
    throw new Error('找不到 SN / Remarks column。');
  }

  const startRow = maxHeaderRow_(cols) + 1;
  const bySn = {};
  const warnings = [];

  for (let r = startRow; r < values.length; r++) {
    const snSource = cleanCell_(values[r][cols.sn]);
    if (!snSource) continue;

    const remarks = cleanCell_(values[r][cols.remarks]);
    const routing = deriveRoutingStatus_(remarks);
    // V7.13: FTZ Schedule GA is fixed at column V (22nd column / index 21).
    const ga = values[r] && values[r].length > 21 ? cleanCell_(values[r][21]) : '';

    const record = {
      snSource: snSource,
      sourceRow: r + 1,
      sourceTab: sheetName || '',
      remarks: remarks,
      routing: routing,
      ga: ga
    };

    const expanded = expandSnRange_(snSource);
    const snList = expanded.length ? expanded : [snSource];

    snList.forEach(function(sn) {
      const key = normalizeSn_(sn);
      if (!key) return;

      if (!bySn[key] || isNewerFtzRecord_(record, bySn[key])) {
        bySn[key] = record;
      }
    });
  }

  return { bySn: bySn, warnings: warnings };
}

function isNewerFtzRecord_(a, b) {
  const ad = ((a || {}).routing || {}).latestDateKey || '';
  const bd = ((b || {}).routing || {}).latestDateKey || '';

  if (ad !== bd) return ad > bd;

  const ar = Number((a || {}).sourceRow || 0);
  const br = Number((b || {}).sourceRow || 0);
  return ar >= br;
}

function emptyRoutingStatus_(stage) {
  return {
    stage: stage || '-',
    tone: 'info',
    latestDate: '',
    latestDateKey: '',
    latestText: '',
    latestAuthor: '',
    authors: [],
    entries: [],
    hasRepeatedGa: false,
    fullRemarks: ''
  };
}

function deriveRoutingStatus_(remarks) {
  const full = cleanCell_(remarks);
  if (!full) {
    return {
      stage: 'NOT STARTED',
      tone: 'info',
      latestDate: '',
      latestDateKey: '',
      latestText: '',
      latestAuthor: '',
      authors: [],
      entries: [],
      hasRepeatedGa: false,
      fullRemarks: ''
    };
  }

  const lines = full
    .split(/\r?\n+/)
    .map(function(x) { return String(x || '').trim(); })
    .filter(Boolean);

  let best = null;

  lines.forEach(function(line, idx) {
    const parsed = parseFtzRemarkLine_(line, idx);
    if (!best) {
      best = parsed;
      return;
    }

    if (parsed.dateKey && !best.dateKey) {
      best = parsed;
      return;
    }

    if (parsed.dateKey && best.dateKey) {
      if (parsed.dateKey > best.dateKey ||
          (parsed.dateKey === best.dateKey && parsed.index > best.index)) {
        best = parsed;
      }
      return;
    }

    if (!parsed.dateKey && !best.dateKey && parsed.index > best.index) {
      best = parsed;
    }
  });

  best = best || { text: full, cleanText: full, dateText: '', dateKey: '', index: 0 };

  const status = classifyRoutingText_(full);

  const parsedEntries = lines.map(function(line, idx) {
    return parseFtzRemarkLine_(line, idx);
  });

  const authors = [];
  parsedEntries.forEach(function(item) {
    const a = String(item.author || '').trim();
    if (a && authors.indexOf(a) === -1) authors.push(a);
  });

  const hasRepeatedGa =
    /repeated\s*ga|repeat\s*ga|ga\s*repeated/i.test(full);

  return {
    stage: status.stage,
    tone: status.tone,
    latestDate: best.dateText || '',
    latestDateKey: best.dateKey || '',
    latestText: best.cleanText || best.text || '',
    latestAuthor: best.author || '',
    authors: authors,
    hasRepeatedGa: hasRepeatedGa,
    entries: parsedEntries.map(function(item) {
      return {
        date: item.dateText || '',
        dateKey: item.dateKey || '',
        author: item.author || '',
        text: item.cleanText || item.text || ''
      };
    }),
    fullRemarks: full
  };
}

function parseFtzRemarkLine_(line, index) {
  const text = String(line || '').trim();

  // Common FTZ formats:
  // 11/6/25 Hoe : RFQ done
  // 28/5/25 Has : GA released. Pending RFQ
  // 20/6/25 Chin : Final released
  const m = text.match(/^\s*(\d{1,2})[\/-](\d{1,2})[\/-](\d{2,4})\b/);

  let dateText = '';
  let dateKey = '';
  let rest = text;

  if (m) {
    const dd = Number(m[1]);
    const mm = Number(m[2]);
    let yy = Number(m[3]);
    if (yy < 100) yy += 2000;

    if (dd >= 1 && dd <= 31 && mm >= 1 && mm <= 12) {
      dateText =
        String(dd).padStart(2, '0') + '-' +
        monthShort_(mm) + '-' +
        String(yy).slice(-2);

      dateKey =
        String(yy) + '-' +
        String(mm).padStart(2, '0') + '-' +
        String(dd).padStart(2, '0');
    }

    rest = text.substring(m[0].length).trim();
  }

  // Capture the updater name immediately before ":".
  // "Has : GA released" -> author "Has", update "GA released"
  let author = '';
  const authorMatch = rest.match(/^([^:]{1,40})\s*:\s*(.*)$/);

  if (authorMatch) {
    author = String(authorMatch[1] || '').trim();
    rest = String(authorMatch[2] || '').trim();
  }

  return {
    text: text,
    cleanText: rest || text,
    author: author,
    dateText: dateText,
    dateKey: dateKey,
    index: index
  };
}

function monthShort_(m) {
  return ['','Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'][Number(m)] || '';
}

function classifyRoutingText_(value) {
  const s = String(value || '').trim();
  const t = s.toLowerCase();

  // Routing workflow:
  // NOT STARTED -> PRELIST -> PENDING RFQ -> FINAL DONE

  if (!t) {
    return { stage: 'NOT STARTED', tone: 'info' };
  }

  // Highest priority completion phrases.

  // RFQ done / complete.
  if (/rfq\s*(done|complete|completed|closed)/.test(t)) {
    return { stage: 'FINAL DONE', tone: 'good' };
  }

  // "RFQ F-part done" also means completed.
  if (/rfq.*f[\s-]*part.*done/.test(t)) {
    return { stage: 'FINAL DONE', tone: 'good' };
  }

  // "GA released. Rechecked done." also means completed.
  if (
    /ga\s*released.*recheck(?:ed)?\s*done/.test(t) ||
    /recheck(?:ed)?\s*done.*ga\s*released/.test(t)
  ) {
    return { stage: 'FINAL DONE', tone: 'good' };
  }

  // "Repeated GA" means routing is completed.
  if (/repeated\s*ga|repeat\s*ga|ga\s*repeated/.test(t)) {
    return { stage: 'FINAL DONE', tone: 'good' };
  }

  // Cancelled orders are treated as completed for Routing.
  if (/order\s*cancel(?:led|ed)?|cancel(?:led|ed)?\s*order/.test(t)) {
    return { stage: 'FINAL DONE', tone: 'good' };
  }

  // Pending RFQ remains an in-process status when no higher-priority
  // completion phrase above is present.
  if (/pending\s*rfq|pending.*rfq|rfq.*pending/.test(t)) {
    return { stage: 'PENDING RFQ', tone: 'warn' };
  }

  // If Final Released exists AND there is no Pending RFQ, it is completed.
  if (/final\s*released|final\s*release/.test(t)) {
    return { stage: 'FINAL DONE', tone: 'good' };
  }

  // Earlier stage.
  if (/pre\s*list|prelist/.test(t)) {
    return { stage: 'PRELIST', tone: 'warn' };
  }

  // Keep this special case if the source explicitly says it.
  if (/not\s*required|not\s*req\b/.test(t)) {
    return { stage: 'NOT REQUIRED', tone: 'good' };
  }

  // A remark exists, but none of the known Routing workflow keywords
  // were found. Do not invent a generic UPDATED state.
  // Keep it at NOT STARTED until a recognized process keyword appears.
  return { stage: 'NOT STARTED', tone: 'warn' };
}


function parseProductionSheets_(sheets, plannerRows, engineerBySn, ftzBySn, ftzConfigured) {
  const masterBySn = {};
  (plannerRows || []).forEach(function(p) {
    masterBySn[normalizeSn_(p.sn)] = p;
  });

  let rows = [];
  let warnings = [];

  sheets.forEach(function(sheet) {
    try {
      const parsed = parseProductionCalendar_(readSheet_(sheet), sheet.getName(), masterBySn);
      rows = rows.concat(parsed.rows);
      warnings = warnings.concat(parsed.warnings || []);
    } catch (e) {
      warnings.push('Production ' + sheet.getName() + ': skipped - ' + e.message);
    }
  });

  // De-duplicate exact same S/N + date entries while preserving first visual sequence.
  const seen = {};
  rows = rows.filter(function(r) {
    const k = normalizeSn_(r.sn) + '|' + (r.productionDateKey || '') + '|' + r.sourceTab;
    if (seen[k]) return false;
    seen[k] = true;
    return true;
  });

  rows.sort(function(a, b) {
    const da = a.productionDateKey || '9999-99-99';
    const db = b.productionDateKey || '9999-99-99';
    if (da !== db) return da < db ? -1 : 1;
    if (a.sourceRow !== b.sourceRow) return a.sourceRow - b.sourceRow;
    return a.sourceCol - b.sourceCol;
  });

  // Diagnostic only: record which production months were found.
  // This makes it easier to detect a future calendar-layout change.
  const monthCoverage = {};
  rows.forEach(function(r) {
    if (r.productionDateKey) {
      const ym = r.productionDateKey.substring(0, 7);
      monthCoverage[ym] = (monthCoverage[ym] || 0) + 1;
    }
  });

  const dailyCounter = {};
  rows.forEach(function(r, i) {
    const master = masterBySn[normalizeSn_(r.sn)] || null;
    const eng = engineerBySn[normalizeSn_(r.sn)] || null;
    const ftz = (ftzBySn || {})[normalizeSn_(r.sn)] || null;

    const dayKey = r.productionDateKey || 'NO-DATE';
    dailyCounter[dayKey] = (dailyCounter[dayKey] || 0) + 1;

    r.productionSeq = i + 1;
    r.dailySeq = dailyCounter[dayKey];

    r.sapSo = master ? master.sapSo : '';
    r.customer = master ? master.customer : '';
    r.endUser = master ? master.endUser : '';
    r.masterModel = master ? master.model : '';
    r.masterWeek = master ? master.plannerWeek : '';
    r.masterYear = master ? master.plannerYear : '';
    r.masterWeekKey = master ? master.weekKey : '';
    r.comDate = master ? master.comDate : '';
    r.comDateKey = master ? master.comDateKey : '';
    r.masterMatched = !!master;

    // Prefer model shown in the Production Planner. Fall back to Masterplan model.
    if (!r.model) r.model = master ? master.model : '';

    r.engineerExempt = isEngineerExemptSn_(r.sn);

    const ftzRoutingFinalFallback =
      !r.engineerExempt &&
      !eng &&
      !!(ftz && ftz.routing && ftz.routing.stage === 'FINAL DONE');

    // Engineer Sheet has priority. FTZ fallback is used only for NO MATCH.
    r.engineerMatched = !!eng || ftzRoutingFinalFallback;
    r.engineerFallbackFromFtz = ftzRoutingFinalFallback;
    r.engineerFallbackSource = ftzRoutingFinalFallback ? 'FTZ ROUTING FINAL DONE' : '';

    // V7.13 Component Allocation: orders on the same Engineer row are one design group.
    // Use the expanded S/N set as the stable group key so formatting / row movement does not split the group.
    const componentGroupSns = eng
      ? expandSnRange_(eng.snSource || r.sn).map(function(x){ return normalizeSnDisplay_(x); })
      : [normalizeSnDisplay_(r.sn)];
    const componentGroupNorm = componentGroupSns
      .map(function(x){ return normalizeSn_(x); })
      .filter(Boolean)
      .sort();
    r.engineerSnSource = eng ? String(eng.snSource || '').trim() : '';
    r.engineerGroupSns = componentGroupSns;
    r.engineerGroupKey = eng && componentGroupNorm.length
      ? ('ENG|' + componentGroupNorm.join(','))
      : ('SN|' + normalizeSn_(r.sn));

    r.mechEngineer = eng ? eng.mechEngineer : '';
    r.elecEngineer = eng ? eng.elecEngineer : '';
    r.mechanical = eng
      ? eng.mechanical
      : (ftzRoutingFinalFallback ? emptyStatus_('FINAL DONE') : emptyStatus_('NO ENGINEER MATCH'));
    r.electrical = eng
      ? eng.electrical
      : (ftzRoutingFinalFallback ? emptyStatus_('FINAL DONE') : emptyStatus_('NO ENGINEER MATCH'));

    r.ftzConfigured = !!ftzConfigured;
    r.ftzMatched = !!ftz;
    r.routing = r.engineerExempt
      ? emptyRoutingStatus_('-')
      : (ftz
        ? ftz.routing
        : emptyRoutingStatus_(ftzConfigured ? 'NOT STARTED' : '-'));
    r.routingRemarks = ftz ? ftz.remarks : '';
    r.routingSourceTab = ftz ? ftz.sourceTab : '';

    // V7.13: Component Allocation GA source is FTZ Schedule column V only.
    // Do not fall back to Production Planner GA values.
    r.ga = ftz ? cleanCell_(ftz.ga) : '';
  });

  if (sheets.length && !rows.length) {
    warnings.push(
      'Production Planner: 没有自动识别到每日排程。当前解析器会寻找 Monday/Tuesday/... + Model + Status 的日历结构，并以 S/N 作为订单识别。'
    );
  }

  return { rows: rows, warnings: warnings };
}


function looksLikeDateText_(value) {
  const s = String(value || '').trim();
  if (!s) return false;

  // 31/12, 13/1, 31-12, 13-1
  if (/^\d{1,2}[\/-]\d{1,2}$/.test(s)) return true;

  // 31/12/2026, 31-12-26
  if (/^\d{1,2}[\/-]\d{1,2}[\/-]\d{2,4}$/.test(s)) return true;

  // 31-Dec, 31-Dec-26, 31 December 2026
  if (/^\d{1,2}[\s-](Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)/i.test(s)) return true;

  return false;
}

// Legacy Production Planner GA lookup retained for parser compatibility.
// V7.13 Component Allocation ignores this value and uses FTZ Schedule column V only.
// Production history remains date-only.
function buildProductionGaLookup_(values) {
  const out = {};

  function isSnHeader_(v) {
    const h = normalizeHeader_(v);
    return h === 'sn' || h === 'serialno' || h === 'serialnumber';
  }
  function isGaHeader_(v) {
    const h = normalizeHeader_(v);
    return h === 'ga' || h === 'gano' || h === 'ganumber' ||
      h === 'gadrawing' || h === 'gadrawingno' || h === 'gadwg' ||
      h === 'generalarrangement' || h === 'generalarrangementdrawing';
  }
  function looksSn_(sn) {
    const k = normalizeSn_(sn);
    return /^[A-Z]?\d{3,7}$/.test(k);
  }
  function add_(snText, gaText) {
    const ga = cleanCell_(gaText);
    if (!ga) return;
    const sns = expandSnRange_(cleanCell_(snText));
    sns.forEach(function(sn) {
      const k = normalizeSn_(sn);
      if (!k || !looksSn_(sn)) return;
      if (!out[k]) out[k] = [];
      if (out[k].indexOf(ga) === -1) out[k].push(ga);
    });
  }

  for (let r = 0; r < values.length; r++) {
    const row = values[r] || [];
    let snCol = -1;
    let gaCol = -1;
    for (let c = 0; c < row.length; c++) {
      if (snCol < 0 && isSnHeader_(row[c])) snCol = c;
      if (gaCol < 0 && isGaHeader_(row[c])) gaCol = c;
    }
    if (snCol < 0 || gaCol < 0 || snCol === gaCol) continue;

    // Read the contiguous lookup table below the detected header. Stop after
    // several blank/non-SN lines so lower unrelated tables are not absorbed.
    let misses = 0;
    for (let rr = r + 1; rr < values.length && misses < 8; rr++) {
      const dataRow = values[rr] || [];
      const snText = cleanCell_(dataRow[snCol]);
      const gaText = cleanCell_(dataRow[gaCol]);
      if (!snText && !gaText) { misses++; continue; }
      const expanded = expandSnRange_(snText);
      if (!expanded.some(looksSn_)) { misses++; continue; }
      misses = 0;
      add_(snText, gaText);
    }
  }

  const compact = {};
  Object.keys(out).forEach(function(k) { compact[k] = out[k].join(' / '); });
  return compact;
}

function parseProductionCalendar_(data, sheetName, masterBySn) {
  const values = data.display;
  const rows = [];
  const warnings = [];

  const months = {
    january:1, february:2, march:3, april:4, may:5, june:6,
    july:7, august:8, september:9, october:10, november:11, december:12
  };

  const weekdays = {
    monday:true, tuesday:true, wednesday:true, thursday:true,
    friday:true, saturday:true, sunday:true
  };

  let currentYear = '';
  let currentMonth = '';
  let groupStarts = [];
  let dayByCol = {};
  let gaOffsetByCol = {};
  const gaBySn = buildProductionGaLookup_(values);

  function rowWords_(row) {
    return row.map(function(v){ return String(v || '').trim().toLowerCase(); });
  }

  for (let r = 0; r < values.length; r++) {
    const row = values[r];
    const words = rowWords_(row);

    // Update year/month context from labels such as "2026" and "September".
    words.forEach(function(w) {
      if (/^20\d{2}$/.test(w)) currentYear = parseInt(w, 10);
      if (months[w]) currentMonth = months[w];
    });

    // Lower auxiliary tables such as "SN / Model / PLAN DATE / STATUS"
    // or "SEPTEMBER TEST PANEL" are not the daily production calendar.
    const compact = words.map(normalizeHeader_);
    const wholeRowText = words.join(' ');
    if (
      (compact.indexOf('sn') !== -1 && compact.indexOf('plandate') !== -1) ||
      compact.indexOf('plandate') !== -1 ||
      /test\s*panel/i.test(wholeRowText)
    ) {
      groupStarts = [];
      dayByCol = {};
      gaOffsetByCol = {};
      continue;
    }

    // Identify the calendar header row:
    // Monday | Model | Status | Tuesday | Model | Status | ...
    const weekdayCols = [];
    words.forEach(function(w, c) {
      if (weekdays[w]) weekdayCols.push(c);
    });

    if (weekdayCols.length >= 2) {
      groupStarts = weekdayCols.slice();
      dayByCol = {};
      gaOffsetByCol = {};
      groupStarts.forEach(function(start, i) {
        const end = i + 1 < groupStarts.length ? groupStarts[i + 1] : Math.min(row.length, start + 6);
        for (let cc = start + 1; cc < end; cc++) {
          const h = normalizeHeader_(row[cc]);
          if (h === 'ga' || h === 'gano' || h === 'ganumber' || h === 'gadrawing' || h === 'gadrawingno' || h === 'gadwg') {
            gaOffsetByCol[start] = cc - start;
            break;
          }
        }
      });
      continue;
    }

    if (!groupStarts.length) continue;

    groupStarts.forEach(function(c) {
      const first = cleanCell_(row[c]);
      const model = c + 1 < row.length ? cleanCell_(row[c + 1]) : '';
      let status = c + 2 < row.length ? cleanCell_(row[c + 2]) : '';
      const gaOffset = gaOffsetByCol[c] || 0;
      let ga = gaOffset && c + gaOffset < row.length ? cleanCell_(row[c + gaOffset]) : '';

      // Date number header (1..31) for this weekday block.
      if (/^\d{1,2}$/.test(first)) {
        const n = parseInt(first, 10);
        if (n >= 1 && n <= 31) {
          dayByCol[c] = n;
          return;
        }
      }

      if (!first || !dayByCol[c]) return;

      // IMPORTANT:
      // Historical production rows often replace the status with an actual
      // completion date such as 7/1, 13/3, 31/7, etc.
      // Those rows are still valid production-calendar entries and MUST be kept.
      // Auxiliary PLAN DATE / TEST PANEL tables are filtered separately above,
      // so do not reject date-like values here.

      const snKey = normalizeSn_(first);
      if (!ga && gaBySn[snKey]) ga = gaBySn[snKey];
      const inMaster = !!masterBySn[snKey];

      // Production planner S/N is normally 4–6 digits; allow letter-prefixed
      // rolling-stock references when Model/Status exists.
      const looksLikeSn =
        /^\d{4,6}$/.test(first) ||
        /^[A-Za-z]\d{4,6}$/.test(first) ||
        /^[A-Za-z]\d{3,7}[A-Za-z0-9-]*$/.test(first);

      if (!inMaster && !(looksLikeSn && (model || status))) return;

      let dateKey = '';
      let dateDisplay = '';
      let prodWeek = '';
      if (currentYear && currentMonth && dayByCol[c]) {
        const d = new Date(currentYear, currentMonth - 1, dayByCol[c]);
        dateKey = Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd');
        dateDisplay = Utilities.formatDate(d, Session.getScriptTimeZone(), 'dd-MMM-yyyy');
        prodWeek = isoWeek_(d);
      }

      rows.push({
        sn: normalizeSnDisplay_(first),
        model: model,
        ga: ga,
        productionStatus: status,
        productionDate: dateDisplay,
        productionDateKey: dateKey,
        productionYear: currentYear || '',
        productionWeek: prodWeek || '',
        productionWeekKey: (currentYear && prodWeek) ? String(currentYear) + '-W' + String(prodWeek).padStart(2, '0') : '',
        sourceTab: sheetName,
        sourceRow: r + 1,
        sourceCol: c + 1
      });
    });
  }

  return { rows: rows, warnings: warnings };
}

function isMasterplanTestPanel_(model) {
  const s = String(model || '')
    .trim()
    .toUpperCase()
    .replace(/[\s_-]+/g, ' ');

  return /\bTEST\s*PANEL\b/.test(s);
}

function parsePlanner_(data) {
  const values = data.display;
  const raw = data.raw;
  const warnings = [];

  const cols = detectColumns_(values, {
    sapSo: ['SAP SO', 'SAPSO', 'SAP SALES ORDER', 'SAP ORDER'],
    sn: ['SN', 'S/N', 'SERIAL NO', 'SERIAL NUMBER'],
    model: ['MODEL'],
    customer: ['CUST.', 'CUST', 'CUSTOMER'],
    endUser: ['END USER', 'ENDUSER'],
    week: ['WEEK', 'WK', 'ORDER WEEK'],
    comDate: ['COM. DATE', 'COM DATE', 'COM.DATE', 'COMMITTED DATE', 'COMMIT DATE']
  }, 20);

  if (cols.sn == null) {
    throw new Error('Planner Sheet 找不到 SN 栏。请确认表头包含 "SN" 或 "S/N"。');
  }
  if (cols.comDate == null) {
    warnings.push('Planner Sheet 未自动找到 Com. Date；日期栏会留空。');
  }

  const headerRow = maxHeaderRow_(cols);
  const rows = [];
  let seq = 0;

  for (let r = headerRow + 1; r < values.length; r++) {
    const sn = cleanCell_(values[r][cols.sn]);
    if (!sn) continue;

    // 跳过明显的说明/汇总行
    if (/^(sn|s\/n)$/i.test(sn)) continue;

    const model = cols.model != null ? cleanCell_(values[r][cols.model]) : '';

    // Masterplan TEST PANEL rows are reference/auxiliary items and should not
    // enter the Production Sequence Monitor order list.
    if (isMasterplanTestPanel_(model)) continue;

    seq++;
    const rawCom = cols.comDate != null ? raw[r][cols.comDate] : '';
    const comDisplay = cols.comDate != null ? cleanCell_(values[r][cols.comDate]) : '';
    const comKey = dateKey_(rawCom, comDisplay);
    const plannerWeekRaw = cols.week != null ? cleanCell_(values[r][cols.week]) : '';
    const yearWeek = buildYearWeek_(plannerWeekRaw, rawCom, comDisplay);

    rows.push({
      seq: seq,
      sapSo: cols.sapSo != null ? cleanCell_(values[r][cols.sapSo]) : '',
      sn: normalizeSnDisplay_(sn),
      model: model,
      customer: cols.customer != null ? cleanCell_(values[r][cols.customer]) : '',
      endUser: cols.endUser != null ? cleanCell_(values[r][cols.endUser]) : '',
      plannerWeek: yearWeek.week,
      plannerYear: yearWeek.year,
      weekKey: yearWeek.key,
      comDate: comDisplay,
      comDateKey: comKey,
      sourceRow: r + 1
    });
  }

  return { rows: rows, warnings: warnings };
}

function parseEngineer_(data) {
  const values = data.display;
  const raw = data.raw;
  const warnings = [];

  const cols = detectColumns_(values, {
    sn: ['S/N', 'SN', 'SERIAL NO', 'SERIAL NUMBER'],
    customer: ['CUSTOMER', 'CUST.', 'CUST'],
    endUser: ['END USER', 'ENDUSER'],
    type: ['TYPE', 'MODEL'],
    qty: ['QTY', 'QUANTITY'],
    salesOrder: ['SALESORDER', 'SALES ORDER'],
    requestedDate: ['REQUESTED DATE'],
    committedDate: ['COMMITTED DATE', 'COMMIT DATE'],
    mechEngineer: ['MECH. ENGINEER', 'MECH ENGINEER', 'MECHANICAL ENGINEER'],
    elecEngineer: ['ELEC. ENGINEER', 'ELEC ENGINEER', 'ELECTRICAL ENGINEER'],

    mechListPlan: ['MECH LIST PLAN', 'MECHANICAL LIST PLAN'],
    mechListActual: ['MECH LIST ACTUAL', 'MECHANICAL LIST ACTUAL'],
    elecListPlan: ['ELEC LIST PLAN', 'ELECTRICAL LIST PLAN'],
    elecListActual: ['ELEC LIST ACTUAL', 'ELECTRICAL LIST ACTUAL'],

    mechArfPlan: ['MECH ARF PLAN', 'MECHANICAL ARF PLAN'],
    mechArfActual: ['MECH ARF ACTUAL', 'MECHANICAL ARF ACTUAL'],
    elecArfPlan: ['ELEC ARF PLAN', 'ELECTRICAL ARF PLAN'],
    elecArfActual: ['ELEC ARF ACTUAL', 'ELECTRICAL ARF ACTUAL'],

    mechFinalPlan: ['MECH FINAL PLAN', 'MECHANICAL FINAL PLAN'],
    mechFinalActual: ['MECH FINAL ACTUAL', 'MECHANICAL FINAL ACTUAL'],
    elecFinalPlan: ['ELEC FINAL PLAN', 'ELECTRICAL FINAL PLAN'],
    elecFinalActual: ['ELEC FINAL ACTUAL', 'ELECTRICAL FINAL ACTUAL']
  }, 20);

  if (cols.sn == null) {
    throw new Error('Engineer Sheet 找不到 S/N 栏。请确认表头包含 "S/N" 或 "SN"。');
  }

  const headerRow = maxHeaderRow_(cols);
  const bySn = {};
  const collisions = [];

  for (let r = headerRow + 1; r < values.length; r++) {
    const snSource = cleanCell_(values[r][cols.sn]);
    if (!snSource) continue;

    const expanded = expandSnRange_(snSource);
    if (!expanded.length) continue;

    const mech = buildEngineeringStatus_(raw[r], values[r], {
      listPlan: cols.mechListPlan,
      listActual: cols.mechListActual,
      arfPlan: cols.mechArfPlan,
      arfActual: cols.mechArfActual,
      finalPlan: cols.mechFinalPlan,
      finalActual: cols.mechFinalActual
    });

    const elec = buildEngineeringStatus_(raw[r], values[r], {
      listPlan: cols.elecListPlan,
      listActual: cols.elecListActual,
      arfPlan: cols.elecArfPlan,
      arfActual: cols.elecArfActual,
      finalPlan: cols.elecFinalPlan,
      finalActual: cols.elecFinalActual
    });

    const committedRaw = cols.committedDate != null ? raw[r][cols.committedDate] : '';
    const committedDisplay = cols.committedDate != null ? cleanCell_(values[r][cols.committedDate]) : '';

    const record = {
      snSource: snSource,
      sourceRow: r + 1,
      customer: cols.customer != null ? cleanCell_(values[r][cols.customer]) : '',
      endUser: cols.endUser != null ? cleanCell_(values[r][cols.endUser]) : '',
      type: cols.type != null ? cleanCell_(values[r][cols.type]) : '',
      qty: cols.qty != null ? cleanCell_(values[r][cols.qty]) : '',
      salesOrder: cols.salesOrder != null ? cleanCell_(values[r][cols.salesOrder]) : '',
      requestedDate: cols.requestedDate != null ? cleanCell_(values[r][cols.requestedDate]) : '',
      committedDate: committedDisplay,
      committedDateKey: dateKey_(committedRaw, committedDisplay),
      mechEngineer: cols.mechEngineer != null ? cleanCell_(values[r][cols.mechEngineer]) : '',
      elecEngineer: cols.elecEngineer != null ? cleanCell_(values[r][cols.elecEngineer]) : '',
      mechanical: mech,
      electrical: elec
    };

    expanded.forEach(function(sn) {
      const key = normalizeSn_(sn);
      if (!key) return;
      if (bySn[key]) {
        collisions.push(key);
      }
      bySn[key] = record;
    });
  }

  if (collisions.length) {
    warnings.push(
      'Engineer Sheet 有重复 S/N 匹配：' +
      Array.from(new Set(collisions)).slice(0, 12).join(', ') +
      (collisions.length > 12 ? ' ...' : '')
    );
  }

  return { bySn: bySn, warnings: warnings };
}

function detectColumns_(values, aliasMap, maxRows) {
  const result = {};
  const meta = {};
  const maxR = Math.min(values.length, maxRows || 20);

  Object.keys(aliasMap).forEach(function(key) {
    const aliases = aliasMap[key].map(normalizeHeader_);
    let found = null;

    for (let r = 0; r < maxR && !found; r++) {
      for (let c = 0; c < values[r].length; c++) {
        const cell = normalizeHeader_(values[r][c]);
        if (!cell) continue;
        if (aliases.indexOf(cell) !== -1) {
          found = { col: c, row: r };
          break;
        }
      }
    }

    result[key] = found ? found.col : null;
    meta[key] = found ? found.row : null;
  });

  result.__headerRows = meta;
  return result;
}

function maxHeaderRow_(cols) {
  const meta = cols.__headerRows || {};
  let max = 0;
  Object.keys(meta).forEach(function(k) {
    if (meta[k] != null && meta[k] > max) max = meta[k];
  });
  return max;
}

function normalizeHeader_(value) {
  return String(value || '')
    .replace(/\n/g, ' ')
    .trim()
    .toLowerCase()
    .replace(/&/g, 'and')
    .replace(/[^a-z0-9]/g, '');
}

function cleanCell_(value) {
  return String(value == null ? '' : value).trim();
}

function normalizeSnDisplay_(value) {
  let s = cleanCell_(value);
  s = s.replace(/\.0$/, '');
  return s;
}

function normalizeSn_(value) {
  return normalizeSnDisplay_(value)
    .toUpperCase()
    .replace(/\s+/g, '');
}

function expandSnRange_(value) {
  const raw = normalizeSnDisplay_(value);
  if (!raw) return [];

  // Engineer 有时会在同一个 S/N cell 输入多个编号，例如：
  // 43375-78, 43895
  // 43375-78 + 43895
  // 43375, 43377 + 43895
  //
  // 先把逗号 / 中文逗号 / + / 分号 / 换行拆成多个片段，
  // 再对每个片段分别处理 range。
  const tokens = raw
    .split(/[,+，;；\n\r]+/)
    .map(function(x) { return String(x || '').trim().replace(/\s+/g, ''); })
    .filter(Boolean);

  const out = [];

  function addUnique_(sn) {
    const s = normalizeSn_(sn);
    if (!s) return;
    if (out.map(normalizeSn_).indexOf(s) === -1) out.push(String(sn));
  }

  function expandOne_(token) {
    // 普通单一 S/N，例如 43895 / L50276
    if (!/^\d+-\d+$/.test(token)) {
      addUnique_(token);
      return;
    }

    // 数字范围，例如：
    // 43375-78   => 43375, 43376, 43377, 43378
    // 44099-44110 => 44099 ... 44110
    const parts = token.split('-');
    const startStr = parts[0];
    const endStr = parts[1];
    const start = parseInt(startStr, 10);
    let end;

    if (endStr.length < startStr.length) {
      const base = Math.pow(10, endStr.length);
      end = Math.floor(start / base) * base + parseInt(endStr, 10);
      while (end < start) end += base;
    } else {
      end = parseInt(endStr, 10);
    }

    // 防止异常输入一次展开太多编号。
    if (!isFinite(start) || !isFinite(end) || end < start || end - start > 500) {
      addUnique_(token);
      return;
    }

    for (let n = start; n <= end; n++) addUnique_(String(n));
  }

  tokens.forEach(expandOne_);
  return out;
}

function buildEngineeringStatus_(rawRow, displayRow, c) {
  function d(col) {
    return col == null ? '' : cleanCell_(displayRow[col]);
  }
  function rv(col) {
    return col == null ? '' : rawRow[col];
  }

  const listPlan = d(c.listPlan);
  const listActual = d(c.listActual);
  const arfPlan = d(c.arfPlan);
  const arfActual = d(c.arfActual);
  const finalPlan = d(c.finalPlan);
  const finalActual = d(c.finalActual);

  let stage = 'NOT STARTED';
  if (listActual) stage = 'LIST DONE';
  if (arfActual) stage = 'ARF DONE';
  if (finalActual) stage = 'FINAL DONE';

  let overdue = false;
  let overdueStep = '';

  if (!finalActual && isPastDate_(rv(c.finalPlan), finalPlan)) {
    overdue = true;
    overdueStep = 'FINAL';
  } else if (!arfActual && isPastDate_(rv(c.arfPlan), arfPlan)) {
    overdue = true;
    overdueStep = 'ARF';
  } else if (!listActual && isPastDate_(rv(c.listPlan), listPlan)) {
    overdue = true;
    overdueStep = 'LIST';
  }

  return {
    stage: stage,
    overdue: overdue,
    overdueStep: overdueStep,
    listPlan: listPlan,
    listActual: listActual,
    arfPlan: arfPlan,
    arfActual: arfActual,
    finalPlan: finalPlan,
    finalActual: finalActual
  };
}

function isEngineerExemptSn_(sn) {
  // G / L prefixed S/N do not require Engineer release status.
  // Normalize first so values such as " G50188", "'G50188", "G 50188"
  // are also detected correctly.
  const normalized = normalizeSn_(sn).replace(/^[^A-Z0-9]+/, '');
  return /^[GL]/.test(normalized);
}

function isPastDate_(rawValue, displayValue) {
  const d = coerceDate_(rawValue, displayValue);
  if (!d) return false;
  const today = new Date();
  today.setHours(0,0,0,0);
  d.setHours(0,0,0,0);
  return d < today;
}

function dateKey_(rawValue, displayValue) {
  const d = coerceDate_(rawValue, displayValue);
  if (!d) return '';
  return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd');
}

function coerceDate_(rawValue, displayValue) {
  if (Object.prototype.toString.call(rawValue) === '[object Date]' && !isNaN(rawValue.getTime())) {
    return new Date(rawValue.getTime());
  }

  const s = String(displayValue || rawValue || '').trim();
  if (!s || s === '-') return null;

  // dd-MMM-yy / dd-MMM-yyyy
  const m = s.match(/^(\d{1,2})[-\/]([A-Za-z]{3})[-\/](\d{2,4})$/);
  if (m) {
    const months = {
      jan:0,feb:1,mar:2,apr:3,may:4,jun:5,
      jul:6,aug:7,sep:8,oct:9,nov:10,dec:11
    };
    const mon = months[m[2].toLowerCase()];
    if (mon != null) {
      let year = parseInt(m[3], 10);
      if (year < 100) year += 2000;
      return new Date(year, mon, parseInt(m[1], 10));
    }
  }

  const parsed = new Date(s);
  return isNaN(parsed.getTime()) ? null : parsed;
}


function buildYearWeek_(plannerWeekRaw, rawComDate, displayComDate) {
  const d = coerceDate_(rawComDate, displayComDate);
  const year = d ? d.getFullYear() : '';
  const week = d ? isoWeek_(d) : '';

  // V5.23:
  // Masterplan Week is derived ONLY from Com. Date.
  // The source Planner "Week" column is intentionally ignored so an incorrect,
  // blank or manually maintained Week value cannot place an order in the wrong
  // group or leave it outside a Week group.
  return {
    year: year || '',
    week: week || '',
    key: (year && week) ? String(year) + '-W' + String(week).padStart(2, '0') : ''
  };
}

function parseWeekNumber_(value) {
  const s = String(value || '').trim();
  if (!s) return '';
  const m = s.match(/(\d{1,2})/);
  if (!m) return '';
  const n = parseInt(m[1], 10);
  return (n >= 1 && n <= 53) ? n : '';
}

function isoWeek_(date) {
  const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
  const dayNum = d.getUTCDay() || 7;
  d.setUTCDate(d.getUTCDate() + 4 - dayNum);
  const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
  return Math.ceil((((d - yearStart) / 86400000) + 1) / 7);
}

function formatDateTime_(d) {
  return Utilities.formatDate(d, Session.getScriptTimeZone(), 'dd-MMM-yyyy HH:mm:ss');
}

function normalizeRoutingSetupParts_(parts) {
  const out = [], seen = {};
  (Array.isArray(parts) ? parts : []).forEach(function(value) {
    const part = String(value || '').toUpperCase().replace(/\s+/g, '').trim();
    if (!/^BR\d{6,}$/.test(part) || seen[part]) return;
    seen[part] = true;
    out.push(part);
  });
  return out.sort();
}

function routingWorkDetailPropertyKey_(groupKey) {
  const raw = Utilities.newBlob(String(groupKey || '')).getBytes();
  return ROUTING_WORK_DETAIL_PROPERTY_PREFIX + Utilities.base64EncodeWebSafe(raw).replace(/=+$/, '').substring(0, 180);
}

function getRoutingWorkDetails(groupKey) {
  const key = String(groupKey || '').trim();
  if (!key) return {};
  const raw = PropertiesService.getUserProperties().getProperty(routingWorkDetailPropertyKey_(key));
  if (!raw) return {};
  try {
    const parsed = JSON.parse(raw);
    return parsed && typeof parsed === 'object' ? parsed : {};
  } catch (e) {
    return {};
  }
}

function saveRoutingWorkDetail(payload) {
  payload = payload || {};
  const groupKey = String(payload.groupKey || '').trim();
  const part = String(payload.part || '').toUpperCase().replace(/\s+/g, '').trim();
  if (!groupKey) throw new Error('Routing group is required.');
  if (!/^BR\d{6,}$/.test(part)) throw new Error('A valid BR number is required.');

  const description = String(payload.description || '').replace(/\s+/g, ' ').trim().substring(0, 160);
  const routeTypeRaw = String(payload.routeType || '').toUpperCase().trim();
  const rfqRaw = String(payload.rfq || '').toUpperCase().replace(/\s+/g, ' ').trim();
  const routeType = routeTypeRaw === 'E' || routeTypeRaw === 'E50' ? routeTypeRaw : '';
  const rfq = rfqRaw === 'RFQ' || rfqRaw === 'NO RFQ' ? rfqRaw : '';
  const propertyKey = routingWorkDetailPropertyKey_(groupKey);
  const lock = LockService.getUserLock();
  lock.waitLock(10000);
  try {
    const properties = PropertiesService.getUserProperties();
    let details = {};
    try { details = JSON.parse(properties.getProperty(propertyKey) || '{}') || {}; } catch (e) { details = {}; }
    if (!description && !routeType && !rfq) delete details[part];
    else details[part] = {
      description: description,
      routeType: routeType,
      rfq: rfq,
      updatedAt: formatDateTime_(new Date())
    };
    properties.setProperty(propertyKey, JSON.stringify(details));
    return {ok:true, part:part, detail:details[part] || null};
  } finally {
    lock.releaseLock();
  }
}

function saveRoutingWorkDone(payload) {
  payload = payload || {};
  const groupKey = String(payload.groupKey || '').trim();
  const phase = String(payload.phase || '').toUpperCase().trim();
  const done = !!payload.done;
  if (!groupKey) throw new Error('Routing group is required.');
  if (phase !== 'PRELIST' && phase !== 'FINAL') throw new Error('A valid Routing phase is required.');

  const propertyKey = routingWorkDetailPropertyKey_(groupKey);
  const lock = LockService.getUserLock();
  lock.waitLock(10000);
  try {
    const properties = PropertiesService.getUserProperties();
    let details = {};
    try { details = JSON.parse(properties.getProperty(propertyKey) || '{}') || {}; } catch (e) { details = {}; }
    if (!details.__phaseDone || typeof details.__phaseDone !== 'object') details.__phaseDone = {};
    details.__phaseDone[phase] = {
      done: done,
      updatedAt: formatDateTime_(new Date())
    };
    properties.setProperty(propertyKey, JSON.stringify(details));
    return {ok:true, phase:phase, done:done, updatedAt:details.__phaseDone[phase].updatedAt};
  } finally {
    lock.releaseLock();
  }
}

function applyRoutingWorkbenchDoneStatuses_(out, groups) {
  (Array.isArray(groups) ? groups : []).forEach(function(g) {
    if (!g || !g.key || !out[g.key]) return;
    const details = getRoutingWorkDetails(g.storageKey || g.key);
    const phaseDone = details && details.__phaseDone;
    if (!phaseDone || typeof phaseDone !== 'object') return;
    ['PRELIST', 'FINAL'].forEach(function(phase) {
      if (!Object.prototype.hasOwnProperty.call(phaseDone, phase)) return;
      const saved = phaseDone[phase] || {};
      out[g.key][phase.toLowerCase()] = saved.done ? {
        status: 'DONE',
        source: 'WORKBENCH',
        updatedAt: String(saved.updatedAt || '')
      } : null;
    });
  });
  return out;
}

function getRoutingSetupStatusMap(groups) {
  const clean = (Array.isArray(groups) ? groups : []).slice(0, 80).map(function(g) {
    const key = String((g || {}).key || '').trim();
    return { key:key, storageKey:String((g || {}).storageKey || key).trim() };
  }).filter(function(g) { return !!g.key; });
  const out = {};
  clean.forEach(function(g) { out[g.key] = { prelist:null, final:null }; });
  if (!clean.length) return out;

  const ss = getOrCreateWorkbenchStore_();
  const sh = ensureRoutingSetupQueueSheet_(ss);
  if (sh.getLastRow() < 2) return applyRoutingWorkbenchDoneStatuses_(out, clean);

  const wanted = {};
  clean.forEach(function(g) {
    if (!wanted[g.storageKey]) wanted[g.storageKey] = [];
    wanted[g.storageKey].push(g.key);
  });
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 10).getValues();
  const batches = {};

  rows.forEach(function(r) {
    const groupKey = String(r[2] || '').trim();
    const phase = String(r[4] || '').toUpperCase();
    const workId = String(r[1] || '').trim();
    if (!wanted[groupKey] || !workId || (phase !== 'PRELIST' && phase !== 'FINAL')) return;
    const key = groupKey + '|' + phase + '|' + workId;
    if (!batches[key]) {
      const ts = r[0] instanceof Date ? r[0] : new Date(r[0]);
      batches[key] = {
        groupKey: groupKey,
        phase: phase,
        workId: workId,
        timestampMs: isNaN(ts.getTime()) ? 0 : ts.getTime(),
        sentAt: isNaN(ts.getTime()) ? '' : formatDateTime_(ts),
        status: String(r[6] || 'WAITING DONE').toUpperCase(),
        parts: [],
        user: String(r[7] || '')
      };
    }
    const part = String(r[5] || '').toUpperCase().trim();
    if (part && batches[key].parts.indexOf(part) < 0) batches[key].parts.push(part);
  });

  Object.keys(batches).forEach(function(key) {
    const x = batches[key];
    const slot = x.phase.toLowerCase();
    x.parts.sort();
    x.partCount = x.parts.length;
    (wanted[x.groupKey] || []).forEach(function(displayKey) {
      const current = out[displayKey][slot];
      if (!current || x.timestampMs > Number(current.timestampMs || 0)) out[displayKey][slot] = x;
    });
  });
  return applyRoutingWorkbenchDoneStatuses_(out, clean);
}

function sendRoutingSetupToSheet(payload) {
  payload = payload || {};
  const groupKey = String(payload.groupKey || '').trim();
  const sns = (Array.isArray(payload.sns) ? payload.sns : [])
    .map(function(sn) { return String(sn || '').trim(); })
    .filter(Boolean);
  const phase = String(payload.phase || '').toUpperCase();
  const parts = normalizeRoutingSetupParts_(payload.parts);
  const sourceSubject = String(payload.sourceSubject || '').trim();
  const sourceMessageId = String(payload.sourceMessageId || '').trim();

  if (!groupKey) throw new Error('Routing group is required.');
  if (!sns.length) throw new Error('At least one S/N is required.');
  if (phase !== 'PRELIST' && phase !== 'FINAL') throw new Error('Phase must be PRELIST or FINAL.');
  if (!parts.length) throw new Error('Select or add at least one valid BR number.');

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const ss = getOrCreateWorkbenchStore_();
    const sh = ensureRoutingSetupQueueSheet_(ss);
    const existing = {};
    if (sh.getLastRow() > 1) {
      sh.getRange(2, 1, sh.getLastRow() - 1, 10).getValues().forEach(function(r) {
        if (String(r[2] || '').trim() !== groupKey) return;
        if (String(r[4] || '').toUpperCase() !== phase) return;
        if (String(r[6] || '').toUpperCase() !== 'WAITING DONE') return;
        const part = String(r[5] || '').toUpperCase().trim();
        if (part) existing[part] = true;
      });
    }

    const now = new Date();
    const workId = Utilities.getUuid();
    const user = workbenchUser_();
    const rows = parts.filter(function(part) { return !existing[part]; }).map(function(part) {
      return [
        now,
        workId,
        groupKey,
        sns.join(', '),
        phase,
        part,
        'WAITING DONE',
        user,
        sourceSubject,
        sourceMessageId
      ];
    });

    if (rows.length) {
      sh.getRange(sh.getLastRow() + 1, 1, rows.length, 10).setValues(rows);
      sh.autoResizeColumn(6);
    }

    return {
      ok: true,
      groupKey: groupKey,
      phase: phase,
      status: 'WAITING DONE',
      inserted: rows.length,
      skipped: parts.length - rows.length,
      parts: parts,
      sentAt: formatDateTime_(now),
      sheetName: sh.getName(),
      sheetUrl: ss.getUrl() + '#gid=' + sh.getSheetId()
    };
  } finally {
    lock.releaseLock();
  }
}

// Save links discovered by the optional browser helper. The helper runs in
// the user's already-authenticated Vault browser session; this function keeps
// the returned BR -> viewer URL mapping reusable for later S/N groups.
function saveVaultDrawingLinks(entries) {
  const list = Array.isArray(entries) ? entries : [];
  const valid = list.map(function(entry) {
    const part = String((entry || {}).br || (entry || {}).part || '')
      .toUpperCase().replace(/\s+/g, '').trim();
    const url = String((entry || {}).url || '').trim();
    if (!/^BR\d{6,}$/.test(part) || !/^https?:\/\//i.test(url)) return null;
    return {
      part: part,
      url: url,
      fileName: String((entry || {}).fileName || '').trim()
    };
  }).filter(Boolean);
  if (!valid.length) return {ok: true, saved: 0};

  const ss = getOrCreateWorkbenchStore_();
  const sh = ensureVaultDrawingLinksSheet_(ss);
  const rowCount = Math.max(0, sh.getLastRow() - 1);
  const rows = rowCount ? sh.getRange(2, 1, rowCount, 4).getValues() : [];
  const rowByPart = {};
  rows.forEach(function(row, index) {
    const part = String(row[0] || '').toUpperCase().replace(/\s+/g, '').trim();
    if (part) rowByPart[part] = index + 2;
  });
  let saved = 0;
  valid.forEach(function(entry) {
    const values = [[entry.part, entry.url, entry.fileName, formatDateTime_(new Date())]];
    if (rowByPart[entry.part]) sh.getRange(rowByPart[entry.part], 1, 1, 4).setValues(values);
    else {
      sh.getRange(sh.getLastRow() + 1, 1, 1, 4).setValues(values);
      rowByPart[entry.part] = sh.getLastRow();
    }
    saved++;
  });
  return {ok: true, saved: saved, sheetUrl: ss.getUrl() + '#gid=' + sh.getSheetId()};
}
